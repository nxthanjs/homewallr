        .from('houses')
        .select('pinned_products')
        .eq('user_id', auth.user.id)
        .maybeSingle()

      if (house && Array.isArray(house.pinned_products)) {
        setPinnedProducts(house.pinned_products)
      }

      const { data: rooms } = await supabase.from('rooms').select('id, name')
      const { data: posts } = await supabase.from('posts').select('room_id, specs')

      if (posts) {
        const extracted: { spec: string; roomName: string }[] = []
        posts.forEach((p) => {
          if (Array.isArray(p.specs)) {
            const rName = rooms?.find((r) => r.id === p.room_id)?.name || 'Whole House'
            p.specs.forEach((s: string) => {
              if (!extracted.some((e) => e.spec.toLowerCase() === s.toLowerCase())) {
                extracted.push({ spec: s, roomName: rName })
              }
            })
          }
        })
        setPostSpecs(extracted)
      }

      setLoading(false)
    }
    load()

  const handleAddSpec = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!user || !form.name.trim()) return

      ...pinnedProducts,
      {
        id: `${Date.now()}`,
        name: form.name.trim(),
        brand: form.brand.trim() || 'House Spec',

          <p className="text-[11px] font-semibold tracking-widest uppercase text-[#9C9A92]">
            PROFILE

              Logbook

              hasStories
                ? 'bg-gradient-to-tr from-[#8FA680] via-[#C6A467] to-[#6D855E] cursor-pointer active:scale-95'

          >
            {avatarUrl ? (
              <img
                src={avatarUrl}

              <div className="w-20 h-20 rounded-full bg-[#263126] text-[#8FA680] flex items-center justify-center text-2xl font-semibold border-2 border-[#151815]">

                {activeStories.length} Story

            <h1 className="text-xl font-semibold text-[#F2EFE9] truncate leading-tight">
              {ownerNames || `@${handle}`}
            </h1>
            <p className="text-xs text-[#9C9A92]">@{handle}</p>
            {bio && <p className="text-xs text-[#D5D2C8] line-clamp-2 pt-0.5">{bio}</p>}

                + Add Spec

                      ×

            </div>
            <button
              type="button"
              onClick={() => setShowPinIdeaModal(true)}
              className="px-3 py-1.5 rounded-full bg-[#8FA680] text-[#121612] text-xs font-semibold"
            >
              + Pin Idea

          {savedPins.length === 0 ? (
            <button

                    ×

                  ) : (
                    <div className="w-full h-24 bg-[#222B22] p-3 flex items-end">

import { NextRequest, NextResponse } from 'next/server'

const MODELS = ['gemini-3.8-flash', 'gemini-3.5-flash-lite']

export async function POST(req: NextRequest) {
  try {
    const apiKey = process.env.GEMINI_API_KEY
    if (!apiKey) {
      return NextResponse.json(
        { error: 'Missing GEMINI_API_KEY in .env.local.' },
        { status: 500 }
      )
    }

    const formData = await req.formData()
    const file = formData.get('file') as File | null

    if (!file) {
      return NextResponse.json({ error: 'No floor plan image uploaded.' }, { status: 400 })
    }

    const arrayBuffer = await file.arrayBuffer()
    const base64Data = Buffer.from(arrayBuffer).toString('base64')
    const mimeType = file.type || 'image/jpeg'

    const svgPrompt = `You are a precision architectural CAD draftsman for HomeWallr. Trace the uploaded floor plan image into an exact, topologically faithful flat top-down 2D architectural blueprint SVG (<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 1000 800">).

CRITICAL GEOMETRY, WALL & TEXT PLACEMENT RULES (WORKS FOR ANY HOUSE LAYOUT):
1. Single Closed Exterior Wall Path:
   - Draw the entire outer boundary of the building as ONE single continuous closed <path> (stroke="#3A3D38" stroke-width="11" stroke-linejoin="miter" fill="none").
   - Never draw duplicate parallel outer walls or floating detached wall blocks outside the building.
   - Keep collinear outer walls on the EXACT same X or Y coordinate (if rooms share a straight top edge or straight bottom edge in the original image, they MUST share the exact same Y coordinate—do not step rooms above or below a straight outer wall).
   - Draw chimney breasts as solid #3A3D38 rectangles attached to the INSIDE face of the wall, projecting inward into the room.
2. Snap All Interior Walls & Extensions:
   - Every interior dividing wall and attached extension (like a porch or outside store) must connect cleanly to the exact centerline of the exterior wall path with zero gaps or overhanging floor fills.
3. Zero Text-Over-Fixture Collisions:
   - NEVER place room titles or dimensions on top of staircases, kitchen hobs, sinks, or toilets.
   - Place each room's label in the open, unobstructed floor area of that room (for example, in a hallway with stairs at one end, place the "Hallway" label in the open floor space away from the stair treads; in a kitchen with a hob at the top edge, place the "Kitchen" label in the clear central floor space below the hob).
   - For small rooms/extensions (like WC, Porch, Store), place the label in a clean rounded white callout box (#FFFFFF stroke="#3A3D38" stroke-width="1") outside the perimeter connected by a thin dotted leader line.
4. Architectural Details & Watermark Removal:
   - Draw crisp staircase treads with a directional arrow, quarter-circle dashed door swing arcs at wall openings, inset double-glazed window breaks along exterior walls, and clean minimal outlines for hobs, sinks, and toilets.
   - COMPLETELY REMOVE all estate agent watermarks (e.g., "MONOPOLY", "BUY SELL RENT"), logos, and floor-level titles (e.g., "Ground Floor", "Building 1").

HOMEWALLR THEME & PALETTE:
- Canvas background: <rect width="1000" height="800" fill="#F5F3EE"/>
- Room floor fills (strictly clipped inside the wall centerlines, with subtle rounded corners rx="4"):
  - Primary living spaces: Soft Sage Green (#C8D3BE or #E7EAE3)
  - Hallways, kitchens, utility/dining spaces: Warm Taupe / Sand (#DFD3C3 or #EAE6DF)
- Typography: Clean sans-serif (font-family="system-ui, -apple-system, sans-serif") in dark charcoal (#2C2E2B) for room titles (font-size="14" font-weight="700") and metric dimensions underneath (font-size="11" fill="#57534E").

Output ONLY the raw <svg>...</svg> markup. Do not wrap in markdown code fences or JSON.`

    const roomsPrompt = `Analyze this floor plan image and extract all visible rooms with their exact names and metric dimensions (width_m, length_m). Return ONLY valid JSON:
{
  "rooms": [
    { "name": "Living Room", "width_m": 3.61, "length_m": 3.73 }
  ]
}`

    const errors: string[] = []
    let svgMarkup: string | null = null

    for (const model of MODELS) {
      try {
        const supportsThinking = model.includes('3.8')
        const response = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [
                {
                  parts: [
                    { text: svgPrompt },
                    { inline_data: { mime_type: mimeType, data: base64Data } },
                  ],
                },
              ],
              generationConfig: {
                temperature: 0.05,
                maxOutputTokens: 8192,
                ...(supportsThinking ? { thinkingConfig: { thinkingBudget: 0 } } : {}),
              },
            }),
          }
        )

        const result = await response.json()

        if (response.ok) {
          let rawText: string = result.candidates?.[0]?.content?.parts?.[0]?.text || ''
          rawText = rawText.replace(/^```(?:svg|xml)?\s*/i, '').replace(/\s*```$/i, '').trim()
          const svgMatch = rawText.match(/<svg[\s\S]*<\/svg>/i)
          if (svgMatch) {
            svgMarkup = svgMatch[0]
            break
          } else {
            const finishReason = result.candidates?.[0]?.finishReason || 'unknown'
            errors.push(`${model}: incomplete SVG (finishReason: ${finishReason})`)
          }
        } else {
          errors.push(`${model}: ${result.error?.message || response.statusText}`)
        }
      } catch (err: unknown) {
        errors.push(`${model}: ${err instanceof Error ? err.message : 'Network error'}`)
      }
    }

    if (!svgMarkup) {
      return NextResponse.json(
        { error: `SVG generation failed:\n${errors.join('\n')}` },
        { status: 500 }
      )
    }

    let rooms: Array<{ name: string; width_m?: number; length_m?: number }> = []
    for (const model of MODELS) {
      try {
        const supportsThinking = model.includes('3.8')
        const response = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [
                {
                  parts: [
                    { text: roomsPrompt },
                    { inline_data: { mime_type: mimeType, data: base64Data } },
                  ],
                },
              ],
              generationConfig: {
                responseMimeType: 'application/json',
                temperature: 0.1,
                ...(supportsThinking ? { thinkingConfig: { thinkingBudget: 0 } } : {}),
              },
            }),
          }
        )

        if (response.ok) {
          const result = await response.json()
          const rawText = result.candidates?.[0]?.content?.parts?.[0]?.text
          if (rawText) {
            const parsed = JSON.parse(rawText)
            if (Array.isArray(parsed.rooms)) {
              rooms = parsed.rooms
              break
            }
          }
        }
      } catch {
        // Non-blocking: SVG already succeeded
      }
    }

    const image2D = `data:image/svg+xml;base64,${Buffer.from(svgMarkup).toString('base64')}`

    return NextResponse.json({
      image2D,
      rooms,
    })
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : 'Unexpected server error'
    return NextResponse.json({ error: message }, { status: 500 })
  }
}

function RoomDetailContent() {

  const [saving, setSaving] = useState(false)

  const [progress, setProgress] = useState(0)
  const [status, setStatus] = useState('Planning')

    if (!roomId || roomId === '[id]') return

    const loadRoom = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')

      const { data: roomData } = await supabase
        .from('rooms')
        .select('*')
        .eq('id', roomId)
        .single()

      if (roomData) {
        setRoom(roomData)
        setProgress(roomData.progress || 0)
        setStatus(roomData.status || 'Planning')
      }

      const { data: postData } = await supabase
        .from('posts')
        .select('*')
        .eq('room_id', roomId)
        .eq('is_story', false)
        .order('created_at', { ascending: false })

      if (postData) setPosts(postData)
      setLoading(false)

  name: string

  const router = useRouter()
  const [user, setUser] = useState<User | null>(null)
  const [rooms, setRooms] = useState<Room[]>([])
  const [loading, setLoading] = useState(true)

  const [roomId, setRoomId] = useState('')
  const [stage, setStage] = useState('Painting & Decorating')

  const [specsInput, setSpecsInput] = useState('')
  const [cost, setCost] = useState('')
  const [contractor, setContractor] = useState('')

  const [previewUrl, setPreviewUrl] = useState<string | null>(null)

  useEffect(() => {
    const init = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')
      setUser(auth.user)

      const { data: roomData } = await supabase
        .from('rooms')
        .select('id, name')
        .order('created_at', { ascending: true })
      if (roomData) {
        setRooms(roomData)
        if (roomData.length > 0) setRoomId(roomData[0].id)
      }
      setLoading(false)
    }
    init()
  }, [router])

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const chosen = e.target.files?.[0] || null
    setFile(chosen)
    if (chosen) {
      setPreviewUrl(URL.createObjectURL(chosen))
    } else {
      setPreviewUrl(null)
    }
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!user) return
    setSubmitting(true)

      mediaType = isVideo ? 'video' : 'image'
      const ext = file.name.split('.').pop() || (isVideo ? 'mp4' : 'jpg')
      const filePath = `${user.id}/${Date.now()}.${ext}`

      const { error: uploadError } = await supabase.storage
        .from('post-images')
        .upload(filePath, file)

        setSubmitting(false)

        room_id: roomId || null,
        stage,
        caption: caption.trim() || null,

        media_type: mediaType,
        is_story: isStory,

        specs: parsedSpecs,
        cost: cost ? Number(cost) : null,
        contractor: contractor.trim() || null,
      },

      alert('Could not save entry: ' + error.message)

          }`}

  useEffect(() => {
    if (plan2dUrl && !img2D) {
      setImg2D(plan2dUrl)
    }
  }, [plan2dUrl, img2D])

  useEffect(() => {
    if (latestPlanUrl && latestPlanUrl !== img2D) {
      setImg2D(latestPlanUrl)
      onRoomsUpdated()
    }
  }, [latestPlanUrl, img2D, onRoomsUpdated])

  const handleUploadAndGenerate = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    const existingNames = rooms.map((r) => r.name)
    if (fileInputRef.current) fileInputRef.current.value = ''
    await startGeneration(file, userId, existingNames)
  }

  const progressPercent = Math.min(92, Math.round((elapsedSeconds / 40) * 92) + 8)

  return (
    <div className="rounded-2xl bg-[#181C18] border border-[#2D352D] p-4 space-y-3.5">
      {/* Top Header & Upload Trigger */}
      <div className="flex items-center justify-between gap-2">
        <div>
          <p className="text-[10px] uppercase tracking-wider font-semibold text-[#C6A467]">
            House Floor Plan
          </p>
          <h3 className="text-sm font-semibold text-[#F2EFE9]">Architectural Blueprint</h3>
        </div>

        <div className="flex items-center gap-2">
          {img2D && !isGenerating && (
            <button
              type="button"
              onClick={() => setFullscreenOpen(true)}
              className="text-[11px] font-medium text-[#8FA680] hover:text-[#A8BF99] underline px-1"
            >
              Expand ⤢
            </button>
          )}

          {showUploadButton && (
            <label className="cursor-pointer inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-[#8FA680] hover:bg-[#9FB690] text-[#121612] text-xs font-semibold transition-colors">
              <span>
                {isGenerating
                  ? `Drafting (${elapsedSeconds}s)...`
                  : img2D
                  ? '↻ Re-scan Plan'
                  : '✨ Generate Plan'}
              </span>
              <input
                ref={fileInputRef}
                type="file"
                accept="image/*"
                disabled={isGenerating}
                onChange={handleUploadAndGenerate}
                className="hidden"
              />
            </label>
          )}
        </div>
      </div>

      {/* Main Generated Floor Plan Vector */}
      {isGenerating ? (
        <div className="rounded-xl border border-[#364436] bg-[#1F271F] py-10 px-5 text-center space-y-3">
          <p className="text-xs font-semibold text-[#F2EFE9]">{stepText}</p>

          <div className="w-full max-w-xs mx-auto h-2 bg-[#121512] rounded-full overflow-hidden border border-[#2F3A2F]">
            <div
              className="h-full bg-[#8FA680] rounded-full transition-all duration-1000"
              style={{ width: `${progressPercent}%` }}
            />
          </div>

          <p className="text-[11px] text-[#9C9A92] leading-relaxed">
            Drafting in the background (~30–60s). You can browse other pages and we&apos;ll notify you as soon as it&apos;s ready.
          </p>
        </div>
      ) : img2D ? (
        <button
          type="button"
          onClick={() => setFullscreenOpen(true)}
          className="block w-full rounded-xl overflow-hidden border border-[#2F382F] bg-[#F5F3EE] p-1 focus:outline-none"
        >
          <img
            src={img2D}
            alt="Styled 2D Architectural Floor Plan"
            className="w-full max-h-80 object-contain mx-auto rounded-lg"
          />
        </button>
      ) : (
        <div className="rounded-xl border border-dashed border-[#495B42] bg-[#141714] p-6 text-center space-y-1.5">
          <p className="text-xs font-medium text-[#F2EFE9]">Upload an estate agent plan or sketch</p>
          <p className="text-[11px] text-[#9C9A92]">
            AI will redraw your exact stairs, chimney breasts, windows, and doors into a clean architectural blueprint (~30–60s).
          </p>
        </div>
      )}

      {/* Quick Room Navigation Pills */}
      {rooms.length > 0 && (
        <div className="flex flex-wrap gap-1.5 pt-1">
          {rooms.map((room) => (
            <Link
              key={room.id}
              href={`/rooms/${room.id}`}
              className="px-2.5 py-1 rounded-full bg-[#1E221E] border border-[#2E352E] hover:border-[#8FA680] text-[11px] text-[#F2EFE9] flex items-center gap-1.5 transition-colors"
            >
              <span>{room.name}</span>
              <span className="text-[10px] font-semibold text-[#8FA680]">{room.progress}%</span>
            </Link>
          ))}
        </div>
      )}

      {/* Full-Screen Modal for Inspecting Details */}
      {fullscreenOpen && img2D && (
        <div
          className="fixed inset-0 z-50 bg-black/90 backdrop-blur-xs flex flex-col justify-between p-4"
          onClick={() => setFullscreenOpen(false)}
        >
          <div className="flex items-center justify-between text-[#F2EFE9]">
            <span className="text-xs font-semibold">2D Architectural Blueprint</span>
            <button
              type="button"
              onClick={() => setFullscreenOpen(false)}
              className="px-3 py-1 rounded-full bg-white/15 text-xs"
            >
              Close ×
            </button>
          </div>

          <div className="my-auto flex items-center justify-center" onClick={(e) => e.stopPropagation()}>
            <img
              src={img2D}
              alt="Full screen floor plan"
              className="max-h-[82vh] max-w-full rounded-2xl object-contain bg-[#F5F3EE] p-1"
            />
          </div>
          <div />
        </div>
      )}
    </div>
  )
}
