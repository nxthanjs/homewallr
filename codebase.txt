'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { supabase } from '@/lib/supabase'
import type { User } from '@supabase/supabase-js'

interface BlockedRecord {
  id: string
  blocked_id: string
  unblocked_at?: string | null
}

export default function SettingsHubPage() {
  const router = useRouter()
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)

  // Privacy & Settings State
  const [isPrivate, setIsPrivate] = useState(false)
  const [savingPrivacy, setSavingPrivacy] = useState(false)
  const [privacyMsg, setPrivacyMsg] = useState<string | null>(null)

  // Blocked Accounts State
  const [blockedList, setBlockedList] = useState<BlockedRecord[]>([])
  const [blockError, setBlockError] = useState<string | null>(null)

  // Password state
  const [newPassword, setNewPassword] = useState('')
  const [confirmPassword, setConfirmPassword] = useState('')
  const [updatingPass, setUpdatingPass] = useState(false)
  const [passMsg, setPassMsg] = useState<{ text: string; isError: boolean } | null>(null)

  useEffect(() => {
    const init = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')
      setUser(auth.user)

      // Fetch House Privacy
      const { data: house } = await supabase
        .from('houses')
        .select('is_private')
        .eq('user_id', auth.user.id)
        .maybeSingle()

      if (house && typeof house.is_private === 'boolean') {
        setIsPrivate(house.is_private)
      }

      // Fetch Blocked Accounts
      const { data: blocks } = await supabase
        .from('blocked_users')
        .select('*')
        .eq('blocker_id', auth.user.id)

      if (blocks) setBlockedList(blocks)

      setLoading(false)
    }
    init()
  }, [router])

  const handleTogglePrivacy = async (val: boolean) => {
    if (!user) return
    setIsPrivate(val)
    setSavingPrivacy(true)
    setPrivacyMsg(null)

    const { error } = await supabase
      .from('houses')
      .upsert({ user_id: user.id, is_private: val }, { onConflict: 'user_id' })

    setSavingPrivacy(false)
    if (error) {
      setPrivacyMsg('Failed to update privacy settings.')
    } else {
      setPrivacyMsg('Privacy settings updated successfully.')
      setTimeout(() => setPrivacyMsg(null), 3000)
    }
  }

  // Unblock Account
  const handleUnblock = async (recordId: string) => {
    const unblockTime = new Date().toISOString()
    const { error } = await supabase
      .from('blocked_users')
      .update({ unblocked_at: unblockTime })
      .eq('id', recordId)

    if (!error) {
      setBlockedList((prev) =>
        prev.map((b) => (b.id === recordId ? { ...b, unblocked_at: unblockTime } : b))
      )
    }
  }

  // Re-block Account with 24h Cooldown Safety Check
  const handleReblock = async (recordId: string, unblockedAt?: string | null) => {
    setBlockError(null)
    if (unblockedAt) {
      const hoursSinceUnblock = (Date.now() - new Date(unblockedAt).getTime()) / (1000 * 60 * 60)
      const COOLDOWN_HOURS = 24
      if (hoursSinceUnblock < COOLDOWN_HOURS) {
        const hoursLeft = Math.ceil(COOLDOWN_HOURS - hoursSinceUnblock)
        setBlockError(`For safety, you must wait ${hoursLeft}h before re-blocking this account.`)
        return
      }
    }

    const { error } = await supabase
      .from('blocked_users')
      .update({ unblocked_at: null })
      .eq('id', recordId)

    if (!error) {
      setBlockedList((prev) =>
        prev.map((b) => (b.id === recordId ? { ...b, unblocked_at: null } : b))
      )
    }
  }

  // Strong Password Validation (Min 10 chars, uppercase, lowercase, number, symbol)
  const handleChangePassword = async (e: React.FormEvent) => {
    e.preventDefault()
    setPassMsg(null)

    const strongPasswordRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[\W_]).{10,}$/

    if (!strongPasswordRegex.test(newPassword)) {
      setPassMsg({
        text: 'Password must be at least 10 characters and include an uppercase letter, lowercase letter, number, and symbol.',
        isError: true,
      })
      return
    }

    if (newPassword !== confirmPassword) {
      setPassMsg({ text: 'New passwords do not match.', isError: true })
      return
    }

    setUpdatingPass(true)
    const { error } = await supabase.auth.updateUser({ password: newPassword })
    setUpdatingPass(false)

    if (error) {
      setPassMsg({ text: error.message, isError: true })
    } else {
      setPassMsg({ text: 'Password updated successfully with strong authentication.', isError: false })
      setNewPassword('')
      setConfirmPassword('')
    }
  }

  const handleSignOut = async () => {
    await supabase.auth.signOut()
    router.push('/login')
    router.refresh()
  }

  if (loading) return <div className="p-6 text-xs text-[#888]">Loading settings...</div>

  return (
    <div className="px-5 pt-6 pb-20 space-y-6 bg-[#111111] min-h-screen max-w-xl mx-auto text-white font-sans">
      <div className="space-y-1">
        <p className="text-[10px] font-bold uppercase tracking-widest text-[#888]">
          Preferences & Account
        </p>
        <h1 className="text-xl font-bold text-white">Settings</h1>
      </div>

      <div className="space-y-5">
        {/* 1. House & Content Shortcuts */}
        <div className="rounded-2xl bg-[#181818] border border-[#262626] overflow-hidden shadow-sm divide-y divide-[#222]">
          <div className="px-4 py-3 bg-[#1C1C1C]">
            <h2 className="text-xs font-bold uppercase tracking-wider text-[#888]">House & Logbook</h2>
          </div>

          <Link href="/timeline" className="flex items-center justify-between px-4 py-3.5 hover:bg-[#1C1C1C] transition-colors">
            <div>
              <p className="text-xs font-bold text-white">View Build Logbook</p>
              <p className="text-[11px] text-[#888]">Review timeline entries, stage progress, and spend</p>
            </div>
            <span className="text-[#888]">→</span>
          </Link>

          <Link href="/shop" className="flex items-center justify-between px-4 py-3.5 hover:bg-[#1C1C1C] transition-colors">
            <div>
              <p className="text-xs font-bold text-white">Edit House Specs & Paints</p>
              <p className="text-[11px] text-[#888]">Manage saved paint codes, tiles, and supplier links</p>
            </div>
            <span className="text-[#888]">→</span>
          </Link>

          <Link href="/floorplan" className="flex items-center justify-between px-4 py-3.5 hover:bg-[#1C1C1C] transition-colors">
            <div>
              <p className="text-xs font-bold text-white">Architectural Floor Plan</p>
              <p className="text-[11px] text-[#888]">View blueprints and room dimensions</p>
            </div>
            <span className="text-[#888]">→</span>
          </Link>
        </div>

        {/* 2. Privacy & Social Settings */}
        <div className="rounded-2xl bg-[#181818] border border-[#262626] p-4 space-y-3.5 shadow-sm">
          <div className="flex items-center justify-between border-b border-[#222] pb-2.5">
            <div>
              <h2 className="text-xs font-bold uppercase tracking-wider text-[#888]">Privacy & Safety</h2>
              <p className="text-xs font-medium text-white pt-0.5">Private House Profile</p>
            </div>
            <button
              type="button"
              onClick={() => handleTogglePrivacy(!isPrivate)}
              disabled={savingPrivacy}
              className={`w-11 h-6 flex items-center rounded-full p-1 transition-colors ${
                isPrivate ? 'bg-cyan-500 justify-end' : 'bg-[#333] justify-start'
              }`}
            >
              <div className="w-4 h-4 rounded-full bg-white shadow-md" />
            </button>
          </div>
          <p className="text-[11px] text-[#888] leading-relaxed">
            When your house profile is private, new visitors must send a follow request to view your room breakdown, photos, and pinned ideas.
          </p>
          {privacyMsg && (
            <p className={`text-[11px] font-medium ${privacyMsg.includes('Failed') ? 'text-red-400' : 'text-cyan-400'}`}>
              {privacyMsg}
            </p>
          )}
        </div>

        {/* 3. Blocked Accounts Manager with 24h Cooldown */}
        <div className="rounded-2xl bg-[#181818] border border-[#262626] p-4 space-y-3 shadow-sm">
          <h2 className="text-xs font-bold uppercase tracking-wider text-[#888]">Blocked Accounts</h2>
          <p className="text-[11px] text-[#888] leading-relaxed">
            Manage restricted accounts. To prevent harassment loops, unblocked accounts carry a 24-hour cooldown before they can be re-blocked.
          </p>

          {blockError && <p className="text-[11px] text-red-400 font-medium">{blockError}</p>}

          {blockedList.length === 0 ? (
            <p className="text-xs text-[#666] italic pt-1">No blocked accounts.</p>
          ) : (
            <div className="space-y-2 pt-1">
              {blockedList.map((item) => {
                const isUnblocked = item.unblocked_at !== null
                return (
                  <div key={item.id} className="flex items-center justify-between p-3 rounded-xl bg-[#111] border border-[#262626]">
                    <div>
                      <p className="text-xs font-bold text-white">User ID: {item.blocked_id.slice(0, 8)}...</p>
                      <p className="text-[10px] text-[#777]">
                        {isUnblocked ? 'Unblocked (cooldown active)' : 'Blocked'}
                      </p>
                    </div>

                    {isUnblocked ? (
                      <button
                        type="button"
                        onClick={() => handleReblock(item.id, item.unblocked_at)}
                        className="px-3.5 py-1.5 rounded-lg bg-[#222] hover:bg-[#2A2A2A] text-xs font-semibold text-[#CCC] border border-[#333]"
                      >
                        Re-block
                      </button>
                    ) : (
                      <button
                        type="button"
                        onClick={() => handleUnblock(item.id)}
                        className="px-3.5 py-1.5 rounded-lg bg-white/10 hover:bg-white/20 text-xs font-semibold text-white"
                      >
                        Unblock
                      </button>
                    )}
                  </div>
                )
              })}
            </div>
          )}
        </div>

        {/* 4. Account Security (Strong Password) */}
        <div className="rounded-2xl bg-[#181818] border border-[#262626] p-4 space-y-3.5 shadow-sm">
          <h2 className="text-xs font-bold uppercase tracking-wider text-[#888]">Account Security</h2>
          <form onSubmit={handleChangePassword} className="space-y-3">
            <div>
              <label className="block text-[11px] font-medium text-[#AAA] mb-1">New Password</label>
              <input
                type="password"
                required
                placeholder="Min 10 chars, uppercase, lowercase, number & symbol"
                value={newPassword}
                onChange={(e) => setNewPassword(e.target.value)}
                className="w-full px-3.5 py-2.5 text-xs bg-[#111] text-white placeholder-[#666] border border-[#262626] rounded-xl focus:outline-none focus:border-white"
              />
            </div>

            <div>
              <label className="block text-[11px] font-medium text-[#AAA] mb-1">Confirm New Password</label>
              <input
                type="password"
                required
                placeholder="Re-enter new password"
                value={confirmPassword}
                onChange={(e) => setConfirmPassword(e.target.value)}
                className="w-full px-3.5 py-2.5 text-xs bg-[#111] text-white placeholder-[#666] border border-[#262626] rounded-xl focus:outline-none focus:border-white"
              />
            </div>

            {passMsg && (
              <p className={`text-[11px] font-medium ${passMsg.isError ? 'text-red-400' : 'text-cyan-400'}`}>
                {passMsg.text}
              </p>
            )}

            <button
              type="submit"
              disabled={updatingPass}
              className="px-4 py-2 bg-white hover:bg-[#E0E0E0] text-black text-xs font-semibold rounded-xl transition-colors"
            >
              {updatingPass ? 'Updating...' : 'Update Password Securely'}
            </button>
          </form>
        </div>

        {/* 5. Sign Out */}
        <div className="pt-2">
          <button
            type="button"
            onClick={handleSignOut}
            className="w-full py-3 rounded-xl bg-[#222] hover:bg-red-950/40 text-red-400 text-xs font-semibold border border-[#333] transition-colors"
          >
            Sign Out of HomeWallr
          </button>
        </div>
      </div>
    </div>
  )
}

'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { supabase } from '@/lib/supabase'
import type { User } from '@supabase/supabase-js'
import type { PinnedProduct } from '@/app/profile/page'

export default function SpecsDirectoryPage() {
  const router = useRouter()
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)

  const [pinnedProducts, setPinnedProducts] = useState<PinnedProduct[]>([])
  const [postSpecs, setPostSpecs] = useState<{ spec: string; roomName: string }[]>([])
  const [search, setSearch] = useState('')

  // Add Spec Modal
  const [showAdd, setShowAdd] = useState(false)
  const [saving, setSaving] = useState(false)
  const [form, setForm] = useState({ name: '', brand: '', price: '', room: '', url: '' })

  useEffect(() => {
    const load = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')
      setUser(auth.user)

      const { data: house } = await supabase
        .from('houses')
        .select('pinned_products')
        .eq('user_id', auth.user.id)
        .maybeSingle()

      if (house && Array.isArray(house.pinned_products)) {
        setPinnedProducts(house.pinned_products)
      }

      const { data: rooms } = await supabase.from('rooms').select('id, name')
      const { data: posts } = await supabase.from('posts').select('room_id, specs')

      if (posts) {
        const extracted: { spec: string; roomName: string }[] = []
        posts.forEach((p) => {
          if (Array.isArray(p.specs)) {
            const rName = rooms?.find((r) => r.id === p.room_id)?.name || 'Whole House'
            p.specs.forEach((s: string) => {
              if (!extracted.some((e) => e.spec.toLowerCase() === s.toLowerCase())) {
                extracted.push({ spec: s, roomName: rName })
              }
            })
          }
        })
        setPostSpecs(extracted)
      }

      setLoading(false)
    }
    load()
  }, [router])

  const handleAddSpec = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!user || !form.name.trim()) return
    setSaving(true)

    const updated: PinnedProduct[] = [
      ...pinnedProducts,
      {
        id: `${Date.now()}`,
        name: form.name.trim(),
        brand: form.brand.trim() || 'House Spec',
        price: form.price.trim() || undefined,
        room: form.room.trim() || undefined,
        url: form.url.trim() || undefined,
      },
    ]

    const { error } = await supabase
      .from('houses')
      .upsert({ user_id: user.id, pinned_products: updated }, { onConflict: 'user_id' })

    setSaving(false)
    if (!error) {
      setPinnedProducts(updated)
      setForm({ name: '', brand: '', price: '', room: '', url: '' })
      setShowAdd(false)
    }
  }

  const handleRemove = async (id: string) => {
    if (!user) return
    const updated = pinnedProducts.filter((p) => p.id !== id)
    setPinnedProducts(updated)
    await supabase.from('houses').upsert({ user_id: user.id, pinned_products: updated }, { onConflict: 'user_id' })
  }

  if (loading) return <div className="p-6 text-xs text-[#888]">Loading house specs...</div>

  const q = search.toLowerCase().trim()
  const filteredPinned = pinnedProducts.filter(
    (p) =>
      !q ||
      p.name.toLowerCase().includes(q) ||
      p.brand.toLowerCase().includes(q) ||
      (p.room || '').toLowerCase().includes(q)
  )
  const filteredTagged = postSpecs.filter(
    (item) => !q || item.spec.toLowerCase().includes(q) || item.roomName.toLowerCase().includes(q)
  )

  return (
    <div className="px-5 pt-6 pb-16 space-y-5 bg-[#111111] min-h-screen max-w-2xl mx-auto">
      <div className="flex items-start justify-between gap-2">
        <div>
          <p className="text-[10px] uppercase tracking-widest font-semibold text-[#888]">
            House Passport Directory
          </p>
          <h1 className="text-lg font-semibold text-white">Specs, Paints & Materials</h1>
          <p className="text-xs text-[#888]">Every paint shade, tile, and fixture in your home</p>
        </div>

        <button
          type="button"
          onClick={() => setShowAdd(true)}
          className="px-4 py-2 rounded-full bg-white hover:bg-[#E0E0E0] text-black text-xs font-semibold shadow-sm transition-colors"
        >
          + Add Spec
        </button>
      </div>

      <input
        type="text"
        placeholder="Search paint shades, brands, rooms..."
        value={search}
        onChange={(e) => setSearch(e.target.value)}
        className="w-full px-4 py-3 text-xs bg-[#181818] text-white placeholder-[#666] border border-[#262626] rounded-xl focus:outline-none focus:border-white"
      />

      {/* Saved / Pinned Product Specs */}
      <div className="space-y-3">
        <h2 className="text-xs font-semibold uppercase tracking-wider text-[#888]">
          Saved House Specs ({filteredPinned.length})
        </h2>

        {filteredPinned.length === 0 ? (
          <div className="rounded-2xl bg-[#181818] border border-[#262626] p-6 text-center space-y-1.5">
            <p className="text-xs font-medium text-white">No saved specs yet</p>
            <p className="text-[11px] text-[#888]">
              Save paint colours, flooring finishes, and supplier links so you never forget a shade or code.
            </p>
          </div>
        ) : (
          <div className="grid grid-cols-2 gap-3">
            {filteredPinned.map((item) => (
              <div
                key={item.id}
                className="rounded-2xl bg-[#181818] border border-[#262626] p-4 space-y-2 relative flex flex-col justify-between shadow-sm"
              >
                <button
                  type="button"
                  onClick={() => handleRemove(item.id)}
                  className="absolute top-2.5 right-3 text-sm text-[#777] hover:text-red-400"
                >
                  ×
                </button>
                <div className="space-y-1">
                  <span className="inline-block text-[10px] px-2 py-0.5 rounded bg-[#222] text-[#CCC] font-semibold">
                    {item.brand}
                  </span>
                  <h3 className="text-xs font-semibold text-white leading-snug">{item.name}</h3>
                  <p className="text-[10px] text-[#888]">
                    {item.room || 'Whole House'}
                    {item.price ? ` · ${item.price}` : ''}
                  </p>
                </div>

                {item.url && (
                  <a
                    href={item.url}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="inline-block text-[11px] font-medium text-white underline pt-1"
                  >
                    Product Link →
                  </a>
                )}
              </div>
            ))}
          </div>
        )}
      </div>

      {/* Auto-Extracted Specs from Logbook Posts */}
      {filteredTagged.length > 0 && (
        <div className="space-y-3">
          <h2 className="text-xs font-semibold uppercase tracking-wider text-[#888]">
            Tagged in Logbook Posts ({filteredTagged.length})
          </h2>
          <div className="flex flex-wrap gap-2">
            {filteredTagged.map((item, idx) => (
              <div
                key={idx}
                className="px-3.5 py-2 rounded-xl bg-[#181818] border border-[#262626] text-xs flex items-center gap-2"
              >
                <span className="text-white font-medium">{item.spec}</span>
                <span className="text-[10px] text-[#888]">{item.roomName}</span>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Add Spec Modal */}
      {showAdd && (
        <div
          className="fixed inset-0 z-50 bg-black/80 backdrop-blur-xs flex items-end sm:items-center justify-center p-4"
          onClick={() => setShowAdd(false)}
        >
          <div
            className="w-full max-w-sm rounded-2xl bg-[#181818] border border-[#333] p-5 space-y-3.5 shadow-2xl"
            onClick={(e) => e.stopPropagation()}
          >
            <div className="flex items-center justify-between">
              <h2 className="text-base font-semibold text-white">Add House Spec</h2>
              <button type="button" onClick={() => setShowAdd(false)} className="text-[#888] text-lg">
                ×
              </button>
            </div>

            <form onSubmit={handleAddSpec} className="space-y-3">
              <input
                type="text"
                required
                placeholder="Item / Paint Name (e.g., Charleston Gray)"

        redirectTo: `${window.location.origin}/settings`,

import { useRouter } from 'next/navigation'
import { supabase } from '@/lib/supabase'

export type WallSide = 'top' | 'bottom' | 'left' | 'right'
export type FeatureType = 'door' | 'window' | 'chimney' | 'stairs'

export interface RoomFeature {
  id: string
  type: FeatureType
  wall: WallSide
  offset: number // percentage along wall (10 to 90)
  size_m?: number // width in meters (e.g. 0.85m door, 1.4m window)
}

export interface PlannerRoom {
  id: string
  name: string
  status: string
  progress: number
  floor_level: string | null
  width_m: number | null
  length_m: number | null
  plan_x: number | null
  plan_y: number | null
  features?: RoomFeature[] | null
}

interface Props {
  rooms: PlannerRoom[]
  onRoomsChange: (rooms: PlannerRoom[]) => void
}

const SCALE = 28 // 28px = 1m
const SNAP = 7   // 7px = 0.25m

const DEFAULT_SIZES: Record<FeatureType, number> = {
  door: 0.85,
  window: 1.2,
  chimney: 1.1,
  stairs: 0.9,
}

export default function FloorPlanBuilder({ rooms, onRoomsChange }: Props) {
  const router = useRouter()
  const canvasRef = useRef<HTMLDivElement>(null)

  const [floor, setFloor] = useState<'Ground Floor' | 'First Floor'>('Ground Floor')
  const [viewMode, setViewMode] = useState<'2d' | '3d'>('2d')
  const [editMode, setEditMode] = useState(false)
  const [zoom, setZoom] = useState(1)

  const [selectedId, setSelectedId] = useState<string | null>(null)
  const [selectedFeatureId, setSelectedFeatureId] = useState<string | null>(null)
  const [activeWall, setActiveWall] = useState<WallSide>('top')

  const [interaction, setInteraction] = useState<{
    type: 'move' | 'resize-right' | 'resize-bottom' | 'resize-left' | 'resize-top'
    roomId: string
    startX: number
    startY: number
    origX: number
    origY: number
    origW: number
    origL: number
  } | null>(null)

  const floorRooms = rooms.filter((r) => (r.floor_level || 'Ground Floor') === floor)
  const selectedRoom = rooms.find((r) => r.id === selectedId) || null
  const selectedFeature = selectedRoom?.features?.find((f) => f.id === selectedFeatureId) || null
  const totalArea = floorRooms.reduce((s, r) => s + (Number(r.width_m) || 3.5) * (Number(r.length_m) || 3), 0)

  const startInteraction = (
    e: React.PointerEvent,
    room: PlannerRoom,
    type: 'move' | 'resize-right' | 'resize-bottom' | 'resize-left' | 'resize-top'
  ) => {
    e.stopPropagation()
    if (selectedId !== room.id) setSelectedFeatureId(null)
    setSelectedId(room.id)
    if (!editMode || viewMode === '3d') return

    setInteraction({
      type,
      roomId: room.id,
      startX: e.clientX,
      startY: e.clientY,
      origX: Number(room.plan_x) || 20,
      origY: Number(room.plan_y) || 20,
      origW: Number(room.width_m) || 3.5,
      origL: Number(room.length_m) || 3.0,
    })
  }

  const handlePointerMove = (e: React.PointerEvent) => {
    if (!interaction) return
    const dx = Math.round((e.clientX - interaction.startX) / zoom / SNAP) * SNAP
    const dy = Math.round((e.clientY - interaction.startY) / zoom / SNAP) * SNAP
    const dMetersX = dx / SCALE
    const dMetersY = dy / SCALE

    onRoomsChange(
      rooms.map((r) => {
        if (r.id !== interaction.roomId) return r
        if (interaction.type === 'move') {
          return { ...r, plan_x: Math.max(0, interaction.origX + dx), plan_y: Math.max(0, interaction.origY + dy) }
        }
        if (interaction.type === 'resize-right') {
          return { ...r, width_m: Math.max(1.25, Number((interaction.origW + dMetersX).toFixed(2))) }
        }
        if (interaction.type === 'resize-bottom') {
          return { ...r, length_m: Math.max(1.25, Number((interaction.origL + dMetersY).toFixed(2))) }
        }
        if (interaction.type === 'resize-left') {
          const newW = Math.max(1.25, Number((interaction.origW - dMetersX).toFixed(2)))
          const actualDx = (interaction.origW - newW) * SCALE
          return { ...r, width_m: newW, plan_x: Math.max(0, interaction.origX + actualDx) }
        }
        if (interaction.type === 'resize-top') {
          const newL = Math.max(1.25, Number((interaction.origL - dMetersY).toFixed(2)))
          const actualDy = (interaction.origL - newL) * SCALE
          return { ...r, length_m: newL, plan_y: Math.max(0, interaction.origY + actualDy) }
        }
        return r
      })
    )
  }

    setSelectedFeatureId(null)

        </div>

        <div className="flex items-center gap-1.5">

            >
              −
            </button>
            <button

              className="px-1 text-[10px] font-medium text-stone-500"
              title="Reset Zoom"
            >
              {Math.round(zoom * 100)}%
            </button>
            <button

              onClick={() => setZoom((z) => Math.min(1.8, Number((z + 0.15).toFixed(2))))}

              title="Zoom In"

              +
            </button>

import { useState, useEffect, createContext, useContext } from 'react'
import { supabase } from '@/lib/supabase'

interface FloorPlanContextType {

  setActiveFloor: (floor: string) => void
}

const FloorPlanContext = createContext<FloorPlanContextType>({
  activeFloor: 'Ground Floor',
  setActiveFloor: () => {},
})

export function FloorPlanProvider({ children }: { children: React.ReactNode }) {
  const [activeFloor, setActiveFloor] = useState('Ground Floor')
  return (
    <FloorPlanContext.Provider value={{ activeFloor, setActiveFloor }}>
      {children}
    </FloorPlanContext.Provider>
  )
}

export function useFloorPlan() {
  return useContext(FloorPlanContext)
}

export function useFloorPlanJob() {
  const [jobStatus, setJobStatus] = useState<string | null>(null)
  const [jobResult, setJobResult] = useState<any>(null)
  const [isGenerating, setIsGenerating] = useState(false)
  const [elapsedSeconds, setElapsedSeconds] = useState(0)
  const [stepText, setStepText] = useState('')
  const [latestPlanUrl, setLatestPlanUrl] = useState<string | null>(null)

  // Animated progress step text & timer
  useEffect(() => {
    let interval: NodeJS.Timeout
    if (isGenerating) {
      setElapsedSeconds(0)
      setStepText('Analyzing architectural layout & walls...')
      interval = setInterval(() => {
        setElapsedSeconds((s) => {
          const next = s + 1
          if (next === 5) setStepText('Tracing exterior boundary & wall centerlines...')
          if (next === 15) setStepText('Detecting doors, windows, stairs & chimney breasts...')
          if (next === 25) setStepText('Extracting room dimensions & generating vector blueprint...')
          if (next === 35) setStepText('Finalizing architectural SVG & saving to House Passport...')
          return next
        })
      }, 1000)
    }
    return () => clearInterval(interval)
  }, [isGenerating])

  // Fully operational AI floor plan generation workflow
  const startGeneration = async (file: File, userId: string, existingNames: string[] = []) => {
    setIsGenerating(true)
    setJobStatus('running')

    try {
      const formData = new FormData()
      formData.append('file', file)

      const res = await fetch('/api/parse-floorplan', {
        method: 'POST',
        body: formData,
      })

      const data = await res.json()
      if (!res.ok) {
        throw new Error(data.error || 'Floor plan parsing failed')
      }

      const { image2D, rooms } = data

      // 1. Save generated 2D plan to houses table
      await supabase
        .from('houses')
        .upsert({ user_id: userId, plan_2d_url: image2D }, { onConflict: 'user_id' })

      // 2. Insert newly detected rooms if they don't already exist
      if (Array.isArray(rooms) && rooms.length > 0) {
        const lowerExisting = new Set(existingNames.map((n) => n.toLowerCase()))
        const newRoomsToInsert = rooms
          .filter((r) => !lowerExisting.has(r.name.toLowerCase()))
          .map((r) => ({
            user_id: userId,
            name: r.name,
            floor_level: 'Ground Floor',
            status: 'Planning',
            progress: 0,
            width_m: r.width_m || null,
            length_m: r.length_m || null,
          }))

        if (newRoomsToInsert.length > 0) {
          await supabase.from('rooms').insert(newRoomsToInsert)
        }
      }

      setLatestPlanUrl(image2D)
      setJobStatus('success')
    } catch (err: any) {
      console.error(err)
      alert('Floor plan generation error: ' + err.message)
      setJobStatus('error')
    } finally {
      setIsGenerating(false)
    }
  }

  return {
    jobStatus,
    setJobStatus,
    jobResult,
    setJobResult,
    isGenerating,
    setIsGenerating,
    elapsedSeconds,
    setElapsedSeconds,
    stepText,
    setStepText,
    latestPlanUrl,
    setLatestPlanUrl,
    startGeneration,
  }
}
