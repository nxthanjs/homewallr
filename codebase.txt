'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { supabase } from '@/lib/supabase'
import type { User } from '@supabase/supabase-js'
import AIFloorPlan from '@/components/AIFloorPlan'

interface RoomCollection {
  id: string
  name: string
  floor_level: string | null
  status: string
  progress: number
  width_m: number | null
  length_m: number | null
}

export default function SettingsPage() {
  const router = useRouter()
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)

  const [savingProfile, setSavingProfile] = useState(false)
  const [uploadingAvatar, setUploadingAvatar] = useState(false)
  const [savingPassword, setSavingPassword] = useState(false)
  const [banner, setBanner] = useState<{ type: 'success' | 'error'; text: string } | null>(null)

  // House & Profile fields (matches your `houses` table)
  const [avatarUrl, setAvatarUrl] = useState<string | null>(null)
  const [plan2dUrl, setPlan2dUrl] = useState<string | null>(null)
  const [ownerNames, setOwnerNames] = useState('')
  const [location, setLocation] = useState('')
  const [bio, setBio] = useState('')
  const [houseTitle, setHouseTitle] = useState('The 1930s Semi')
  const [houseEra, setHouseEra] = useState('1930s Semi-Detached')
  const [rooms, setRooms] = useState<RoomCollection[]>([])

  // Password Change fields
  const [newPassword, setNewPassword] = useState('')
  const [confirmPassword, setConfirmPassword] = useState('')

  const fetchRooms = async () => {
    const { data: roomData } = await supabase
      .from('rooms')
      .select('id, name, floor_level, status, progress, width_m, length_m')
      .order('created_at', { ascending: true })
    if (roomData) setRooms(roomData)
  }

  useEffect(() => {
    const loadSettings = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) {
        router.push('/login')
        return
      }
      setUser(auth.user)

      const { data: house } = await supabase
        .from('houses')
        .select('*')
        .eq('user_id', auth.user.id)
        .maybeSingle()

      if (house) {
        if (house.avatar_url) setAvatarUrl(house.avatar_url)
        if (house.plan_2d_url) setPlan2dUrl(house.plan_2d_url)
        setOwnerNames(house.owner_names || '')
        setLocation(house.location || '')
        setBio(house.bio || '')
        if (house.title) setHouseTitle(house.title)
        if (house.era) setHouseEra(house.era)
      }

      await fetchRooms()
      setLoading(false)
    }

    loadSettings()
  }, [router])

  const handleAvatarUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file || !user) return
    setUploadingAvatar(true)
    setBanner(null)

    const ext = file.name.split('.').pop()
    const path = `avatars/${user.id}-${Date.now()}.${ext}`
    const { error: uploadErr } = await supabase.storage.from('post-images').upload(path, file)
    if (uploadErr) {
      setBanner({ type: 'error', text: 'Photo upload failed: ' + uploadErr.message })
      setUploadingAvatar(false)
      return
    }

    const { data } = supabase.storage.from('post-images').getPublicUrl(path)
    await supabase.from('houses').upsert({ user_id: user.id, avatar_url: data.publicUrl }, { onConflict: 'user_id' })
    setAvatarUrl(data.publicUrl)
    setUploadingAvatar(false)
    setBanner({ type: 'success', text: 'Profile photo updated.' })
  }

  const handleSaveProfile = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!user) return
    setSavingProfile(true)
    setBanner(null)

    const { error } = await supabase.from('houses').upsert(
      {
        user_id: user.id,
        avatar_url: avatarUrl,
        owner_names: ownerNames.trim() || null,
        location: location.trim() || null,
        bio: bio.trim() || null,
        title: houseTitle.trim() || 'The 1930s Semi',
        era: houseEra.trim() || '1930s Semi-Detached',
      },
      { onConflict: 'user_id' }
    )

    setSavingProfile(false)
    if (error) {
      setBanner({ type: 'error', text: 'Error saving settings: ' + error.message })
    } else {
      setBanner({ type: 'success', text: 'Profile & house settings saved.' })
    }
  }

  const handleUpdatePassword = async (e: React.FormEvent) => {
    e.preventDefault()
    setBanner(null)

    if (newPassword.length < 8) {
      setBanner({ type: 'error', text: 'Password must be at least 8 characters.' })
      return
    }
    if (newPassword !== confirmPassword) {
      setBanner({ type: 'error', text: 'Passwords do not match.' })
      return
    }

    setSavingPassword(true)
    const { error } = await supabase.auth.updateUser({ password: newPassword })
    setSavingPassword(false)

    if (error) {
      setBanner({ type: 'error', text: error.message })
    } else {
      setNewPassword('')
      setConfirmPassword('')
      setBanner({ type: 'success', text: 'Account password updated successfully.' })
    }
  }

  const handleSignOut = async () => {
    await supabase.auth.signOut()
    router.push('/login')
  }

  if (loading) return <div className="p-6 text-xs text-stone-500">Loading settings...</div>

  return (
    <div className="pb-14 bg-[#F5F3EE] min-h-screen text-[#2C2E2B]">
      <div className="max-w-xl mx-auto px-5 pt-5 space-y-6">
        {/* Header */}
        <div className="flex items-center justify-between">
          <div>
            <p className="text-[11px] font-semibold tracking-widest uppercase text-[#5C6E52]">
              ACCOUNT & HOUSE
            </p>
            <h1 className="text-xl font-semibold text-stone-900 mt-0.5">Settings</h1>
          </div>
          <button
            type="button"
            onClick={() => router.push('/profile')}
            className="px-3.5 py-1.5 rounded-full text-xs font-medium bg-white border border-[#DCD6CA] text-stone-800 hover:bg-[#EAE6DF] transition-colors"
          >
            ← Profile
          </button>
        </div>

        {/* Status Banner */}
        {banner && (
          <div
            className={`p-3.5 rounded-2xl text-xs font-medium border ${
              banner.type === 'success'
                ? 'bg-[#E7EAE3] border-[#7A8F6D] text-[#1E261B]'
                : 'bg-red-50 border-red-200 text-red-800'
            }`}
          >
            {banner.text}
          </div>
        )}

        {/* 1. Profile & House Identity */}
        <form
          onSubmit={handleSaveProfile}
          className="rounded-2xl bg-white border border-[#DCD6CA] p-5 space-y-4"
        >
          <div>
            <h2 className="text-sm font-semibold text-stone-900">Profile & House Details</h2>
            <p className="text-[11px] text-stone-500">
              Update your profile info, location, and house passport title.
            </p>
          </div>

          <div className="flex items-center justify-between bg-[#FAF8F4] p-3 rounded-xl border border-[#E5E0D5]">
            <div className="flex items-center gap-3">
              {avatarUrl ? (
                <img src={avatarUrl} alt="Avatar" className="w-10 h-10 rounded-full object-cover" />
              ) : (
                <div className="w-10 h-10 rounded-full bg-[#8FA282] text-white flex items-center justify-center text-sm font-semibold">
                  {(ownerNames || user?.email || 'H')[0].toUpperCase()}
                </div>
              )}
              <span className="text-xs font-medium text-stone-800">Profile Photo</span>

            <label className="text-xs font-semibold text-[#4B5E42] underline cursor-pointer">

'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { supabase } from '@/lib/supabase'
import type { PinnedProduct } from '@/app/profile/page'

export default function ShopPage() {
  const router = useRouter()
  const [pinned, setPinned] = useState<PinnedProduct[]>([])
  const [loggedSpecs, setLoggedSpecs] = useState<string[]>([])
  const [search, setSearch] = useState('')
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const loadShop = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')

      const { data: house } = await supabase
        .from('houses')
        .select('pinned_products')
        .eq('user_id', auth.user.id)
        .maybeSingle()

      if (house && Array.isArray(house.pinned_products)) {
        setPinned(house.pinned_products)
      }

      const { data: posts } = await supabase.from('posts').select('specs')
      if (posts) {
        setLoggedSpecs(Array.from(new Set(posts.flatMap((p) => p.specs || []))))
      }

      setLoading(false)
    }
    loadShop()
  }, [router])

  const filteredPinned = pinned.filter(
    (p) =>
      p.name.toLowerCase().includes(search.toLowerCase()) ||
      p.brand.toLowerCase().includes(search.toLowerCase()) ||
      (p.room || '').toLowerCase().includes(search.toLowerCase())
  )

  const filteredSpecs = loggedSpecs.filter((s) => s.toLowerCase().includes(search.toLowerCase()))

  if (loading) return <div className="p-6 text-xs text-stone-500">Loading House Shop...</div>

  return (
    <div className="p-5 space-y-6">
      <div className="space-y-1">
        <p className="text-xs tracking-wider text-[#5C6E52] uppercase font-semibold">HomeWallr Shop</p>
        <h1 className="text-xl font-semibold text-stone-900">House Spec & Product Directory</h1>
        <p className="text-xs text-stone-500">Every paint shade, fixture, and pinned product used across your home.</p>
      </div>

      {/* Search Bar */}
      <input
        type="text"
        placeholder="Search paints, brands, rooms..."
        value={search}
        onChange={(e) => setSearch(e.target.value)}
        className="w-full px-3.5 py-2.5 text-sm bg-white border border-[#D8D3C8] rounded-xl text-stone-900"
      />

      {/* Curated / Pinned Products Grid */}
      <div className="space-y-3">
        <div className="flex justify-between items-center">
          <h2 className="text-xs tracking-wider uppercase font-semibold text-stone-700">Pinned Products ({filteredPinned.length})</h2>
          <Link href="/profile" className="text-xs text-[#5C6E52] font-medium">+ Pin More</Link>
        </div>

        {filteredPinned.length === 0 ? (
          <div className="p-5 rounded-2xl bg-[#EAE6DF] border border-[#DCD6CA] text-xs text-stone-600 space-y-2">
            <p>No pinned products yet. Pin your favourite paints, brassware, tiles, or lighting from your Profile to build your shoppable house directory.</p>
            <Link href="/profile" className="inline-block font-semibold text-[#3B4D32] underline">
              Go to Profile to Pin Products →
            </Link>
          </div>
        ) : (
          <div className="grid grid-cols-2 gap-3">
            {filteredPinned.map((item) => (
              <div key={item.id} className="p-3.5 rounded-2xl bg-white border border-[#DCD6CA] flex flex-col justify-between space-y-3">
                <div className="space-y-1">
                  <span className="inline-block text-[10px] font-medium px-2 py-0.5 rounded bg-[#DCE6D5] text-[#2E3B27]">
                    {item.brand}
                  </span>
                  <p className="text-xs font-semibold text-stone-900 leading-snug">{item.name}</p>
                  {item.room && <p className="text-[11px] text-stone-500">{item.room}</p>}
                </div>
                <div className="flex items-center justify-between pt-2 border-t border-[#EAE6DF]">
                  <span className="text-xs font-semibold text-stone-900">{item.price || '—'}</span>
                  {item.url && (
                    <a href={item.url} target="_blank" rel="noopener noreferrer" className="text-[11px] font-semibold text-[#5C6E52] underline">
                      View →
                    </a>
                  )}
                </div>
              </div>
            ))}
          </div>
        )}
      </div>

      {/* Logged Renovation Specs */}
      {filteredSpecs.length > 0 && (
        <div className="space-y-2.5">
          <h2 className="text-xs tracking-wider uppercase font-semibold text-stone-700">Logged Materials & Paints</h2>
          <div className="space-y-2">
            {filteredSpecs.map((spec, i) => (
              <div key={i} className="p-3 rounded-xl bg-white border border-[#DCD6CA] flex justify-between items-center text-xs">
                <span className="font-medium text-stone-800">{spec}</span>
                <span className="text-[10px] text-stone-400">Logged Spec</span>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  )
}

      setLoading(false)
    }
    loadProfileData()
  }, [router])

  const handleCreateIdea = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!user || !newIdea.title.trim()) return
    setPinningIdea(true)

    let imageUrl: string | null = null
    if (ideaFile) {
      const ext = ideaFile.name.split('.').pop() || 'jpg'
      const path = `ideas/${user.id}-${Date.now()}.${ext}`
      const { error: upErr } = await supabase.storage.from('post-images').upload(path, ideaFile)
      if (!upErr) {
        const { data } = supabase.storage.from('post-images').getPublicUrl(path)
        imageUrl = data.publicUrl
      }
    }

    const { data, error } = await supabase
      .from('saved_pins')
      .insert([
        {
          user_id: user.id,
          title: newIdea.title.trim(),
          category: newIdea.category,
          image_url: imageUrl,
          source_url: newIdea.source_url.trim() || null,
          notes: newIdea.notes.trim() || null,
        },
      ])
      .select()
      .single()

    setPinningIdea(false)
    if (error) {
      alert('Error pinning idea: ' + error.message)
    } else if (data) {
      setSavedPins((prev) => [data, ...prev])
      setNewIdea({ title: '', category: 'Interior Look', source_url: '', notes: '' })
      setIdeaFile(null)
      setShowPinIdeaModal(false)
    }
  }

  const handleDeleteIdea = async (id: string) => {
    const { error } = await supabase.from('saved_pins').delete().eq('id', id)
    if (!error) setSavedPins((prev) => prev.filter((p) => p.id !== id))
  }

  const handleAddPinnedProduct = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!user || !newProduct.name.trim()) return
    setPinningProduct(true)

    const updated: PinnedProduct[] = [
      ...pinnedProducts,
      {
        id: `${Date.now()}`,
        name: newProduct.name.trim(),
        brand: newProduct.brand.trim() || 'House Spec',
        price: newProduct.price.trim() || undefined,
        room: newProduct.room.trim() || undefined,
        url: newProduct.url.trim() || undefined,
      },
    ]

    const { error } = await supabase
      .from('houses')
      .upsert({ user_id: user.id, pinned_products: updated }, { onConflict: 'user_id' })

    setPinningProduct(false)
    if (!error) {
      setPinnedProducts(updated)
      setNewProduct({ name: '', brand: '', price: '', room: '', url: '' })
      setShowPinProductModal(false)
    }
  }

  const removePinnedProduct = async (id: string) => {
    if (!user) return
    const updated = pinnedProducts.filter((p) => p.id !== id)
    setPinnedProducts(updated)
    await supabase.from('houses').upsert({ user_id: user.id, pinned_products: updated }, { onConflict: 'user_id' })
  }

  if (loading) return <div className="p-6 text-xs text-stone-500">Loading profile...</div>

  const handle = user?.email?.split('@')[0] || 'homeowner'
  const displayInitial = (ownerNames || handle)[0].toUpperCase()
  const avgProgress = rooms.length > 0 ? Math.round(rooms.reduce((a, r) => a + r.progress, 0) / rooms.length) : 0
  const hasStories = activeStories.length > 0

  const getRoomName = (roomId: string | null) => {
    if (!roomId) return 'Whole House'
    return rooms.find((r) => r.id === roomId)?.name || 'Room'
  }

  const visiblePosts =
    selectedRoomFilter === 'all'
      ? gridPosts
      : gridPosts.filter((p) => p.room_id === selectedRoomFilter)

  return (
    <div className="pb-12 bg-[#F5F3EE] min-h-screen">
      {/* 1. Profile Identity Section (Single Global Header in layout.tsx sits above this) */}
      <div className="px-5 pt-4 pb-3 space-y-3">
        <div className="flex items-center justify-between">
          <p className="text-[11px] font-semibold tracking-widest uppercase text-stone-700">
            PROFILE
          </p>
          <div className="flex items-center gap-2">
            <Link
              href="/timeline"
              className="text-[11px] font-medium text-[#2C2E2B] bg-[#E7EAE3] hover:bg-[#D7DFD0] border border-[#CFD9C8] px-3 py-1 rounded-full transition-colors"
            >
              Timeline
            </Link>
            <Link
              href="/settings"
              className="text-[11px] font-medium text-[#2C2E2B] bg-white hover:bg-[#EAE6DF] border border-[#DCD6CA] px-3 py-1 rounded-full transition-colors"
            >
              Edit Profile
            </Link>
          </div>
        </div>

        <div className="flex items-center gap-4">
          <button
            type="button"
            onClick={() => hasStories && setStoryIndex(0)}
            className={`relative shrink-0 rounded-full p-0.5 transition-transform ${
              hasStories
                ? 'bg-gradient-to-tr from-[#6F8462] via-[#A9B99D] to-[#C88A6B] cursor-pointer active:scale-95'
                : 'bg-transparent cursor-default'
            }`}
          >
            {avatarUrl ? (
              <img
                src={avatarUrl}
                alt={ownerNames || handle}
                className="w-20 h-20 rounded-full object-cover border-2 border-[#F5F3EE]"
              />
            ) : (
              <div className="w-20 h-20 rounded-full bg-[#8FA282] text-white flex items-center justify-center text-2xl font-semibold border-2 border-[#F5F3EE]">
                {displayInitial}
              </div>
            )}
            {hasStories && (
              <span className="absolute -bottom-1 left-1/2 -translate-x-1/2 bg-[#3A3D38] text-white text-[9px] font-semibold px-1.5 py-0.5 rounded-full border border-[#F5F3EE]">
                {activeStories.length} Story
              </span>
            )}
          </button>

          <div className="space-y-0.5 min-w-0">
            <h1 className="text-xl font-semibold text-stone-900 truncate leading-tight">
              {ownerNames || `@${handle}`}
            </h1>
            <p className="text-xs text-stone-600">@{handle}</p>
            {bio && <p className="text-xs text-stone-700 line-clamp-2 pt-0.5">{bio}</p>}
            {location && <p className="text-xs font-medium text-stone-800 underline pt-0.5">{location}</p>}
          </div>
        </div>

        {/* Stats Row */}
        <div className="flex items-center justify-between pt-1 text-xs text-stone-800 border-t border-[#E6E1D6]">
          <span>
            Houses: <strong className="font-semibold">1</strong>
          </span>
          <span>
            Rooms: <strong className="font-semibold">{rooms.length}</strong>
          </span>
          <span>
            Posts: <strong className="font-semibold">{gridPosts.length}</strong>
          </span>
        </div>
      </div>

      {/* 2. Full-Width Sage-Tinted Milestone Band */}
      <div className="bg-[#E3E9DF] border-y border-[#CFD9C8] px-5 py-3.5 space-y-2.5">
        <div className="flex items-center justify-between">
          <span className="text-xs font-semibold text-[#1E261B]">Renovation Stage</span>
          <span className="text-[11px] font-semibold text-[#4B5E42]">{avgProgress}% Complete</span>
        </div>

        <div className="relative pt-1">
          <div className="w-full h-2 bg-white/80 rounded-full overflow-hidden">
            <div
              className="h-full bg-[#7A8F6D] rounded-full transition-all duration-500"
              style={{ width: `${Math.max(12, avgProgress)}%` }}
            />
          </div>
          <div className="flex justify-between text-[10px] font-medium text-stone-700 pt-1.5">
            <span className={avgProgress >= 0 ? 'text-[#263320] font-semibold' : ''}>Planning</span>
            <span className={avgProgress >= 40 ? 'text-[#263320] font-semibold' : ''}>Structural</span>
            <span className={avgProgress >= 80 ? 'text-[#263320] font-semibold' : ''}>Interior Style</span>
          </div>
        </div>
      </div>

      {/* 3. Clean Underlined Tabs */}
      <div className="px-5 pt-2 border-b border-[#DCD6CA] flex items-center justify-between bg-[#F5F3EE]">
        <button
          type="button"
          onClick={() => setActiveTab('house')}
          className={`pb-2.5 text-xs font-semibold transition-colors border-b-2 ${
            activeTab === 'house'
              ? 'border-[#3A3D38] text-stone-950'
              : 'border-transparent text-stone-500 hover:text-stone-800'
          }`}
        >
          My House
        </button>

        <button
          type="button"
          onClick={() => setActiveTab('posts')}
          className={`pb-2.5 text-xs font-semibold transition-colors border-b-2 ${
            activeTab === 'posts'
              ? 'border-[#3A3D38] text-stone-950'
              : 'border-transparent text-stone-500 hover:text-stone-800'
          }`}
        >
          Posts ({gridPosts.length})
        </button>

        <button
          type="button"
          onClick={() => setActiveTab('pinned_ideas')}
          className={`pb-2.5 text-xs font-semibold transition-colors border-b-2 ${
            activeTab === 'pinned_ideas'
              ? 'border-[#3A3D38] text-stone-950'
              : 'border-transparent text-stone-500 hover:text-stone-800'
          }`}
        >
          Pinned Ideas ({savedPins.length})
        </button>
      </div>

      {/* TAB 1: MY HOUSE */}
      {activeTab === 'house' && (
        <div className="px-5 pt-4 space-y-4">
          <div className="rounded-2xl bg-[#E7EAE3] border border-[#D2DCCB] p-4 space-y-3">
            <div className="flex items-start justify-between gap-2">
              <div>
                <h2 className="text-base font-semibold text-stone-900">{houseTitle}</h2>
                <p className="text-[11px] text-stone-600">{houseEra}</p>
              </div>
              <span className="px-2.5 py-1 rounded-full bg-white/80 text-[11px] font-semibold text-[#3B4D32]">
                {avgProgress}% Complete
              </span>
            </div>

            {user && (
              <AIFloorPlan
                userId={user.id}
                rooms={rooms}
                plan2dUrl={plan2dUrl}
                onRoomsUpdated={fetchRooms}
              />
            )}
          </div>

          <div className="rounded-2xl bg-white border border-[#DCD6CA] p-4 space-y-2.5">
            <div className="flex items-center justify-between">
              <h3 className="text-xs font-semibold uppercase tracking-wider text-stone-700">
                House Specs & Paints ({pinnedProducts.length})
              </h3>
              <button
                type="button"
                onClick={() => setShowPinProductModal(true)}
                className="text-xs font-medium text-[#4B5E42] underline"
              >
                + Add Spec
              </button>
            </div>

            {pinnedProducts.length === 0 ? (
              <p className="text-xs text-stone-500">
                Save your paint colours, flooring, and hardware specs here for easy reference.
              </p>
            ) : (
              <div className="grid grid-cols-2 gap-2">
                {pinnedProducts.map((item) => (
                  <div
                    key={item.id}
                    className="p-2.5 rounded-xl bg-[#FAF8F4] border border-[#E5E0D5] space-y-1 relative"
                  >
                    <button
                      type="button"
                      onClick={() => removePinnedProduct(item.id)}
                      className="absolute top-1.5 right-2 text-xs text-stone-400 hover:text-red-700"
                    >
                      ×
                    </button>
                    <span className="inline-block text-[9px] px-1.5 py-0.5 rounded bg-[#E7EAE3] text-[#2E3B27] font-medium">
                      {item.brand}
                    </span>
                    <p className="text-xs font-semibold text-stone-900 line-clamp-1">{item.name}</p>
                    <p className="text-[10px] text-stone-500">
                      {item.price || 'Saved'} {item.room ? `· ${item.room}` : ''}
                    </p>
                  </div>
                ))}
              </div>
            )}
          </div>
        </div>
      )}

      {/* TAB 2: POSTS GRID */}
      {activeTab === 'posts' && (
        <div className="pt-3 space-y-3">
          {rooms.length > 0 && (
            <div className="px-5 flex items-center gap-1.5 overflow-x-auto no-scrollbar">
              <button
                type="button"
                onClick={() => setSelectedRoomFilter('all')}
                className={`shrink-0 px-3 py-1 rounded-full text-[11px] font-medium ${
                  selectedRoomFilter === 'all'
                    ? 'bg-[#3A3D38] text-white'
                    : 'bg-white text-stone-700 border border-[#DCD6CA]'
                }`}
              >
                All Rooms
              </button>
              {rooms.map((room) => (
                <button
                  key={room.id}
                  type="button"
                  onClick={() => setSelectedRoomFilter(room.id)}
                  className={`shrink-0 px-3 py-1 rounded-full text-[11px] font-medium ${
                    selectedRoomFilter === room.id
                      ? 'bg-[#7A8F6D] text-white'
                      : 'bg-white text-stone-700 border border-[#DCD6CA]'
                  }`}
                >
                  {room.name}
                </button>
              ))}
            </div>
          )}

          {visiblePosts.length === 0 ? (
            <div className="mx-5 p-8 rounded-2xl bg-[#EAE6DF] border border-[#DCD6CA] text-center space-y-2">
              <p className="text-xs text-stone-700">No photos or videos posted yet.</p>
              <Link
                href="/upload"
                className="inline-block px-4 py-2 rounded-full bg-[#7A8F6D] text-white text-xs font-medium"
              >
                + Post First Update
              </Link>
            </div>
          ) : (
            <div className="grid grid-cols-3 gap-0.5 bg-[#DCD6CA]">
              {visiblePosts.map((post) => {
                const isVideo = post.media_type === 'video'
                const hasSpecs = Array.isArray(post.specs) && post.specs.length > 0
                return (
                  <button
                    key={post.id}
                    type="button"
                    onClick={() => setSelectedPost(post)}
                    className="relative aspect-square bg-[#EAE6DF] overflow-hidden group text-left focus:outline-none"
                  >
                    {post.image_url ? (
                      isVideo ? (
                        <video
                          src={post.image_url}
                          muted
                          playsInline
                          className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-200"
                        />
                      ) : (
                        <img
                          src={post.image_url}
                          alt={post.caption || 'Renovation photo'}
                          className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-200"
                        />
                      )
                    ) : (
                      <div className="w-full h-full p-2.5 flex flex-col justify-between bg-[#E7EAE3]">
                        <span className="text-[9px] uppercase tracking-wider font-semibold text-[#5C6E52]">
                          {getRoomName(post.room_id)}
                        </span>
                        <p className="text-[11px] text-stone-800 line-clamp-4 leading-snug">
                          {post.caption || post.stage || 'Note'}
                        </p>
                      </div>
                    )}

                    {isVideo && (
                      <span className="absolute top-1.5 right-1.5 w-5 h-5 rounded-full bg-stone-900/70 text-white flex items-center justify-center text-[9px]">
                        ▶
                      </span>
                    )}

                    {!isVideo && hasSpecs && (
                      <span className="absolute top-1.5 right-1.5 w-5 h-5 rounded-full bg-stone-900/65 text-white flex items-center justify-center text-[10px]">
                        🏷
                      </span>
                    )}

                    {selectedRoomFilter === 'all' && post.room_id && post.image_url && (
                      <span className="absolute bottom-1.5 left-1.5 max-w-[85%] truncate bg-stone-900/65 text-white text-[9px] font-medium px-1.5 py-0.5 rounded">
                        {getRoomName(post.room_id)}
                      </span>
                    )}
                  </button>
                )
              })}
            </div>
          )}
        </div>
      )}

      {/* TAB 3: PINNED IDEAS MOODBOARD */}
      {activeTab === 'pinned_ideas' && (
        <div className="px-5 pt-4 space-y-3">
          <div className="flex items-center justify-between">
            <div>
              <h2 className="text-sm font-semibold text-stone-900">Pinned Ideas & Looks</h2>
              <p className="text-[11px] text-stone-500">Saved inspiration from the Discover feed & web</p>
            </div>
            <button
              type="button"
              onClick={() => setShowPinIdeaModal(true)}
              className="px-3 py-1.5 rounded-full bg-[#7A8F6D] text-white text-xs font-medium"
            >
              + Pin Idea
            </button>
          </div>

          {savedPins.length === 0 ? (
            <button
              type="button"
              onClick={() => setShowPinIdeaModal(true)}
              className="w-full p-6 rounded-2xl bg-[#E5EAE0] border border-dashed border-[#8FA282] text-xs text-stone-800 text-center space-y-1"
            >
              <p className="font-semibold">Start your Pinned Ideas board</p>
              <p className="text-[11px] text-stone-600">
                Save kitchen layouts, paint shades, or garden inspiration you love.
              </p>
            </button>
          ) : (
            <div className="grid grid-cols-2 gap-3">
              {savedPins.map((pin) => (
                <div
                  key={pin.id}
                  className="rounded-2xl bg-white border border-[#DCD6CA] overflow-hidden flex flex-col justify-between relative"
                >
                  <button
                    type="button"
                    onClick={() => handleDeleteIdea(pin.id)}
                    title="Remove pin"
                    className="absolute top-2 right-2 z-10 w-6 h-6 rounded-full bg-stone-900/65 text-white text-xs flex items-center justify-center"
                  >
                    ×
                  </button>

                  {pin.image_url ? (
                    <img src={pin.image_url} alt={pin.title} className="w-full h-36 object-cover bg-[#EAE6DF]" />
                  ) : (
                    <div className="w-full h-24 bg-[#E7EAE3] p-3 flex items-end">
                      <span className="text-[10px] font-semibold uppercase tracking-wider text-[#5C6E52]">
                        {pin.category || 'Idea'}
                      </span>
                    </div>
                  )}

                  <div className="p-3 space-y-1.5 flex-1 flex flex-col justify-between">
                    <div>
                      {pin.image_url && pin.category && (
                        <span className="text-[9px] font-semibold uppercase tracking-wider text-[#5C6E52]">
                          {pin.category}
                        </span>
                      )}
                      <p className="text-xs font-semibold text-stone-900 line-clamp-2">{pin.title}</p>
                      {pin.notes && <p className="text-[11px] text-stone-600 line-clamp-2 mt-0.5">{pin.notes}</p>}
                    </div>

                    {pin.source_url && (
                      <a
                        href={pin.source_url}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="inline-block text-[11px] font-medium text-[#4B5E42] underline pt-1"
                      >
                        Open Link →
                      </a>
                    )}
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {/* Lightbox Modal */}
      {selectedPost && (
        <div
          className="fixed inset-0 z-50 bg-stone-950/80 backdrop-blur-xs flex items-center justify-center p-4"
          onClick={() => setSelectedPost(null)}
        >
          <div
            className="w-full max-w-sm rounded-2xl bg-[#F5F3EE] overflow-hidden shadow-2xl space-y-3"
            onClick={(e) => e.stopPropagation()}
          >
            <div className="px-4 py-3 flex items-center justify-between border-b border-[#DCD6CA]">
              <div>
                <p className="text-xs font-semibold text-stone-900">{getRoomName(selectedPost.room_id)}</p>
                {selectedPost.stage && (
                  <p className="text-[10px] text-[#5C6E52] font-medium">{selectedPost.stage}</p>
                )}
              </div>
              <div className="flex items-center gap-3">
                <Link href="/timeline" className="text-xs font-medium text-stone-700 underline">
                  Edit in Timeline
                </Link>
                <button
                  type="button"
                  onClick={() => setSelectedPost(null)}
                  className="w-7 h-7 rounded-full bg-[#EAE6DF] text-stone-800 flex items-center justify-center text-sm"
                >
                  ×
                </button>
              </div>
            </div>

            {selectedPost.image_url && (
              <div className="bg-stone-900">
                {selectedPost.media_type === 'video' ? (
                  <video src={selectedPost.image_url} controls autoPlay playsInline className="w-full max-h-96" />
                ) : (
                  <img src={selectedPost.image_url} alt="" className="w-full max-h-96 object-contain mx-auto" />
                )}
              </div>
            )}

            <div className="px-4 pb-4 space-y-2">
              {selectedPost.caption && <p className="text-xs text-stone-800 leading-relaxed">{selectedPost.caption}</p>}
              {selectedPost.specs && selectedPost.specs.length > 0 && (
                <div className="flex flex-wrap gap-1.5 pt-1">
                  {selectedPost.specs.map((s, i) => (
                    <span key={i} className="text-[10px] px-2 py-0.5 rounded bg-[#DCE6D5] text-[#2E3B27] font-medium">
                      {s}
                    </span>
                  ))}
                </div>
              )}
            </div>
          </div>
        </div>

            storyIndex + 1 < activeStories.length ? setStoryIndex(storyIndex + 1) : setStoryIndex(null)

            <div className="flex gap-1">

                  <div className={`h-full bg-white ${idx <= storyIndex ? 'w-full' : 'w-0'}`} />

                type="button"

                  e.stopPropagation()
                  setStoryIndex(null)

              >
                Close ×

              activeStories[storyIndex].media_type === 'video' ? (

import { NextRequest, NextResponse } from 'next/server'

const MODELS = ['gemini-3.8-flash', 'gemini-3.5-flash-lite']

export async function POST(req: NextRequest) {
  try {
    const apiKey = process.env.GEMINI_API_KEY
    if (!apiKey) {
      return NextResponse.json(
        { error: 'Missing GEMINI_API_KEY in .env.local.' },
        { status: 500 }
      )
    }

    const formData = await req.formData()
    const file = formData.get('file') as File | null

    if (!file) {
      return NextResponse.json({ error: 'No floor plan image uploaded.' }, { status: 400 })
    }

    const arrayBuffer = await file.arrayBuffer()
    const base64Data = Buffer.from(arrayBuffer).toString('base64')
    const mimeType = file.type || 'image/jpeg'

    const svgPrompt = `You are a precision architectural CAD draftsman for HomeWallr. Trace the uploaded floor plan image into an exact, topologically faithful flat top-down 2D architectural blueprint SVG (<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 1000 800">).

CRITICAL GEOMETRY, WALL & TEXT PLACEMENT RULES (WORKS FOR ANY HOUSE LAYOUT):
1. Single Closed Exterior Wall Path:
   - Draw the entire outer boundary of the building as ONE single continuous closed <path> (stroke="#3A3D38" stroke-width="11" stroke-linejoin="miter" fill="none").
   - Never draw duplicate parallel outer walls or floating detached wall blocks outside the building.
   - Keep collinear outer walls on the EXACT same X or Y coordinate (if rooms share a straight top edge or straight bottom edge in the original image, they MUST share the exact same Y coordinate—do not step rooms above or below a straight outer wall).
   - Draw chimney breasts as solid #3A3D38 rectangles attached to the INSIDE face of the wall, projecting inward into the room.
2. Snap All Interior Walls & Extensions:
   - Every interior dividing wall and attached extension (like a porch or outside store) must connect cleanly to the exact centerline of the exterior wall path with zero gaps or overhanging floor fills.
3. Zero Text-Over-Fixture Collisions:
   - NEVER place room titles or dimensions on top of staircases, kitchen hobs, sinks, or toilets.
   - Place each room's label in the open, unobstructed floor area of that room (for example, in a hallway with stairs at one end, place the "Hallway" label in the open floor space away from the stair treads; in a kitchen with a hob at the top edge, place the "Kitchen" label in the clear central floor space below the hob).
   - For small rooms/extensions (like WC, Porch, Store), place the label in a clean rounded white callout box (#FFFFFF stroke="#3A3D38" stroke-width="1") outside the perimeter connected by a thin dotted leader line.
4. Architectural Details & Watermark Removal:
   - Draw crisp staircase treads with a directional arrow, quarter-circle dashed door swing arcs at wall openings, inset double-glazed window breaks along exterior walls, and clean minimal outlines for hobs, sinks, and toilets.
   - COMPLETELY REMOVE all estate agent watermarks (e.g., "MONOPOLY", "BUY SELL RENT"), logos, and floor-level titles (e.g., "Ground Floor", "Building 1").

HOMEWALLR THEME & PALETTE:
- Canvas background: <rect width="1000" height="800" fill="#F5F3EE"/>
- Room floor fills (strictly clipped inside the wall centerlines, with subtle rounded corners rx="4"):
  - Primary living spaces: Soft Sage Green (#C8D3BE or #E7EAE3)
  - Hallways, kitchens, utility/dining spaces: Warm Taupe / Sand (#DFD3C3 or #EAE6DF)
- Typography: Clean sans-serif (font-family="system-ui, -apple-system, sans-serif") in dark charcoal (#2C2E2B) for room titles (font-size="14" font-weight="700") and metric dimensions underneath (font-size="11" fill="#57534E").

Output ONLY the raw <svg>...</svg> markup. Do not wrap in markdown code fences or JSON.`

    const roomsPrompt = `Analyze this floor plan image and extract all visible rooms with their exact names and metric dimensions (width_m, length_m). Return ONLY valid JSON:
{
  "rooms": [
    { "name": "Living Room", "width_m": 3.61, "length_m": 3.73 }
  ]
}`

    const errors: string[] = []
    let svgMarkup: string | null = null

    for (const model of MODELS) {
      try {
        const supportsThinking = model.includes('3.8')
        const response = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [
                {
                  parts: [
                    { text: svgPrompt },
                    { inline_data: { mime_type: mimeType, data: base64Data } },
                  ],
                },
              ],
              generationConfig: {
                temperature: 0.05,
                maxOutputTokens: 8192,
                ...(supportsThinking ? { thinkingConfig: { thinkingBudget: 0 } } : {}),
              },
            }),
          }
        )

        const result = await response.json()

        if (response.ok) {
          let rawText: string = result.candidates?.[0]?.content?.parts?.[0]?.text || ''
          rawText = rawText.replace(/^```(?:svg|xml)?\s*/i, '').replace(/\s*```$/i, '').trim()
          const svgMatch = rawText.match(/<svg[\s\S]*<\/svg>/i)
          if (svgMatch) {
            svgMarkup = svgMatch[0]
            break
          } else {
            const finishReason = result.candidates?.[0]?.finishReason || 'unknown'
            errors.push(`${model}: incomplete SVG (finishReason: ${finishReason})`)
          }
        } else {
          errors.push(`${model}: ${result.error?.message || response.statusText}`)
        }
      } catch (err: unknown) {
        errors.push(`${model}: ${err instanceof Error ? err.message : 'Network error'}`)
      }
    }

    if (!svgMarkup) {
      return NextResponse.json(
        { error: `SVG generation failed:\n${errors.join('\n')}` },
        { status: 500 }
      )
    }

    let rooms: Array<{ name: string; width_m?: number; length_m?: number }> = []
    for (const model of MODELS) {
      try {
        const supportsThinking = model.includes('3.8')
        const response = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [
                {
                  parts: [
                    { text: roomsPrompt },
                    { inline_data: { mime_type: mimeType, data: base64Data } },
                  ],
                },
              ],
              generationConfig: {
                responseMimeType: 'application/json',
                temperature: 0.1,
                ...(supportsThinking ? { thinkingConfig: { thinkingBudget: 0 } } : {}),
              },
            }),
          }
        )

        if (response.ok) {
          const result = await response.json()
          const rawText = result.candidates?.[0]?.content?.parts?.[0]?.text
          if (rawText) {
            const parsed = JSON.parse(rawText)
            if (Array.isArray(parsed.rooms)) {
              rooms = parsed.rooms
              break
            }
          }
        }
      } catch {
        // Non-blocking: SVG already succeeded
      }
    }

    const image2D = `data:image/svg+xml;base64,${Buffer.from(svgMarkup).toString('base64')}`

    return NextResponse.json({
      image2D,
      rooms,
    })
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : 'Unexpected server error'
    return NextResponse.json({ error: message }, { status: 500 })
  }
}

  progress: number
  notes: string | null

  name: string

  id: string

  caption: string | null
  image_url: string | null

  'Furnishing & Styling',
  'Snagging & Maintenance',
]

export default function RoomDetailPage() {

  const router = useRouter()
  const roomId = params?.id as string

  const [room, setRoom] = useState<RoomDetail | null>(null)
  const [allRooms, setAllRooms] = useState<RoomOption[]>([])
  const [posts, setPosts] = useState<PostItem[]>([])
  const [loading, setLoading] = useState(true)

  // Edit Room state
  const [editingRoom, setEditingRoom] = useState(false)
  const [roomForm, setRoomForm] = useState({
    name: '',
    floor_level: 'Ground Floor',
    width_m: 3.5,
    length_m: 3.0,
    notes: '',
  })
  const [savingRoom, setSavingRoom] = useState(false)

  // Edit Post state
  const [editingPost, setEditingPost] = useState<PostItem | null>(null)
  const [postForm, setPostForm] = useState({
    room_id: '',
    stage: MILESTONE_STAGES[0],
    caption: '',
    specsInput: '',
    cost: '',
    contractor: '',
  })
  const [savingPost, setSavingPost] = useState(false)

  useEffect(() => {
    const loadRoom = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')

      const { data: roomData } = await supabase
        .from('rooms')
        .select('*')
        .eq('id', roomId)
        .single()

      if (roomData) {
        setRoom(roomData)
        setRoomForm({
          name: roomData.name || '',
          floor_level: roomData.floor_level || 'Ground Floor',
          width_m: roomData.width_m || 3.5,
          length_m: roomData.length_m || 3.0,
          notes: roomData.notes || '',
        })
      }

      const { data: roomsList } = await supabase
        .from('rooms')
        .select('id, name')
        .order('created_at', { ascending: true })
      if (roomsList) setAllRooms(roomsList)

      const { data: postData } = await supabase
        .from('posts')
        .select('*')
        .eq('room_id', roomId)
        .order('created_at', { ascending: false })
      if (postData) setPosts(postData)

      setLoading(false)
    }
    if (roomId) loadRoom()
  }, [roomId, router])

  const handleSaveRoom = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!room) return
    setSavingRoom(true)

    const { data, error } = await supabase
      .from('rooms')
      .update({
        name: roomForm.name.trim() || room.name,
        floor_level: roomForm.floor_level,
        width_m: Number(roomForm.width_m),
        length_m: Number(roomForm.length_m),
        notes: roomForm.notes.trim() || null,
      })
      .eq('id', room.id)
      .select()
      .single()

    setSavingRoom(false)
    if (error) {
      alert('Error updating room: ' + error.message)
    } else if (data) {
      setRoom(data)
      setAllRooms((prev) => prev.map((r) => (r.id === data.id ? { id: data.id, name: data.name } : r)))
      setEditingRoom(false)
    }

    if (!room) return
    if (!confirm(`Delete "${room.name}"? Any posts inside will become unassigned rather than deleted.`)) return

    await supabase.from('posts').update({ room_id: null }).eq('room_id', room.id)
    const { error } = await supabase.from('rooms').delete().eq('id', room.id)
    if (error) alert('Error deleting room: ' + error.message)
    else router.push('/')
  }

  const handleProgressChange = async (newProgress: number) => {
    if (!room) return
    const newStatus = newProgress === 100 ? 'Completed' : newProgress === 0 ? 'Planned' : 'In Progress'
    setRoom({ ...room, progress: newProgress, status: newStatus })
    await supabase.from('rooms').update({ progress: newProgress, status: newStatus }).eq('id', room.id)
  }

    setPostForm({
      room_id: post.room_id || '',
      stage: post.stage || MILESTONE_STAGES[0],
      caption: post.caption || '',
      specsInput: Array.isArray(post.specs) ? post.specs.join(', ') : '',
      cost: post.cost !== null && post.cost !== undefined ? String(post.cost) : '',
      contractor: post.contractor || '',
    })
  }

  const handleSavePost = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!editingPost) return
    setSavingPost(true)

    const parsedSpecs = postForm.specsInput
      .split(',')
      .map((s) => s.trim())
      .filter(Boolean)

    const updates = {
      room_id: postForm.room_id || null,
      stage: postForm.stage,
      caption: postForm.caption.trim() || null,
      specs: parsedSpecs.length > 0 ? parsedSpecs : null,
      cost: postForm.cost !== '' ? Number(postForm.cost) : null,
      contractor: postForm.contractor.trim() || null,
    }

    const { data, error } = await supabase
      .from('posts')
      .update(updates)
      .eq('id', editingPost.id)
      .select()
      .single()

    setSavingPost(false)
    if (error) {
      alert('Error updating post: ' + error.message)
    } else if (data) {
      // If moved to a different room, remove from this room's list; otherwise update in place
      if (data.room_id !== roomId) {
        setPosts((prev) => prev.filter((p) => p.id !== data.id))
      } else {
        setPosts((prev) => prev.map((p) => (p.id === data.id ? data : p)))
      }
      setEditingPost(null)
    }
  }

  const handleDeletePost = async () => {
    if (!editingPost) return
    if (!confirm('Delete this post permanently?')) return

    const { error } = await supabase.from('posts').delete().eq('id', editingPost.id)
    if (error) {
      alert('Error deleting post: ' + error.message)
    } else {
      setPosts((prev) => prev.filter((p) => p.id !== editingPost.id))
      setEditingPost(null)
    }
  }

  if (loading) return <div className="p-6 text-xs text-stone-500">Loading room...</div>
  if (!room) return <div className="p-6 text-xs text-stone-600">Room not found.</div>

  const totalRoomSpend = posts.reduce((sum, p) => sum + (Number(p.cost) || 0), 0)
  const roomSpecs = Array.from(new Set(posts.flatMap((p) => p.specs || [])))

  return (
    <div className="p-5 space-y-5">
      {/* Top Navigation */}
      <div className="flex items-center justify-between text-xs">
        <Link href="/" className="font-medium text-stone-600 hover:text-stone-950">
          ← Back to Passport
        </Link>
        <button
          onClick={() => setEditingRoom(!editingRoom)}
          className="font-medium text-[#4B5E42] underline"
        >
          {editingRoom ? 'Close Editor' : 'Edit Room'}
        </button>
      </div>

      {/* Collapsible Edit Room Card */}
      {editingRoom && (
        <form
          onSubmit={handleSaveRoom}
          className="p-4 rounded-2xl bg-[#EAE6DF] border border-[#DCD6CA] space-y-3"
        >
          <div className="flex items-center justify-between">
            <p className="text-xs font-semibold uppercase tracking-wider text-stone-800">Edit Room Details</p>
            <button
              type="button"
              onClick={handleDeleteRoom}
              className="text-[11px] text-red-700 font-medium hover:underline"
            >
              Delete Room
            </button>
          </div>

          <input
            type="text"
            required
            placeholder="Room Name"
            value={roomForm.name}
            onChange={(e) => setRoomForm({ ...roomForm, name: e.target.value })}
            className="w-full px-3 py-2 text-sm bg-white border border-[#D8D3C8] rounded-xl"
          />

          <div className="grid grid-cols-3 gap-2">
            <select
              value={roomForm.floor_level}
              onChange={(e) => setRoomForm({ ...roomForm, floor_level: e.target.value })}
              className="px-2 py-2 text-xs bg-white border border-[#D8D3C8] rounded-xl"

              <option value="Ground Floor">Ground</option>
              <option value="First Floor">First</option>
              <option value="Attic / Loft">Attic / Loft</option>
              <option value="Exterior / Garden">Exterior</option>
            </select>

              type="number"
              step="0.1"
              placeholder="Width (m)"
              value={roomForm.width_m}
              onChange={(e) => setRoomForm({ ...roomForm, width_m: Number(e.target.value) })}
              className="px-2.5 py-2 text-xs bg-white border border-[#D8D3C8] rounded-xl"
            />
            <input
              type="number"
              step="0.1"
              placeholder="Length (m)"
              value={roomForm.length_m}
              onChange={(e) => setRoomForm({ ...roomForm, length_m: Number(e.target.value) })}
              className="px-2.5 py-2 text-xs bg-white border border-[#D8D3C8] rounded-xl"
            />
          </div>

          <textarea
            rows={2}
            placeholder="Scope notes..."
            value={roomForm.notes}
            onChange={(e) => setRoomForm({ ...roomForm, notes: e.target.value })}
            className="w-full px-3 py-2 text-xs bg-white border border-[#D8D3C8] rounded-xl resize-none"
          />

          <button
            type="submit"
            disabled={savingRoom}
            className="w-full py-2.5 bg-[#7A8F6D] text-white text-xs font-medium rounded-xl"
          >
            {savingRoom ? 'Saving Room...' : 'Save Room Changes'}
          </button>
        </form>
      )}

            <h1 className="text-xl font-semibold text-stone-900">{room.name}</h1>

              <div className="flex gap-2 pt-2">
                <button
                  type="button"
                  onClick={handleDeletePost}
                  className="px-3 py-2.5 bg-red-50 text-red-700 border border-red-200 text-xs font-medium rounded-xl"

                  {savingPost ? 'Saving...' : 'Save Changes'}
                </button>

            </form>
          </div>
        </div>
      )}
    </div>
  )

'use client'

import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { supabase } from '@/lib/supabase'

interface RoomOption {
  id: string
  name: string
}

interface PostItem {
  id: string
  room_id: string | null
  stage: string | null
  caption: string | null
  image_url: string | null
  specs: string[] | null
  cost: number | null
  contractor: string | null
  created_at: string
}

const MILESTONE_STAGES = [
  'As It Was (Got the Keys)',
  'Thinking & Planning',
  'Strip-Out & Prep',
  'Structural & First Fix',
  'Plastering & Surfaces',
  'Second Fix & Joinery',
  'Decorating & Painting',
  'Furnishing & Styling',
  'Snagging & Maintenance',
]

export default function TimelinePage() {
  const router = useRouter()
  const [posts, setPosts] = useState<PostItem[]>([])
  const [rooms, setRooms] = useState<RoomOption[]>([])
  const [selectedRoomFilter, setSelectedRoomFilter] = useState<string>('all')
  const [loading, setLoading] = useState(true)

  // Edit Post Modal state
  const [editingPost, setEditingPost] = useState<PostItem | null>(null)
  const [editForm, setEditForm] = useState({
    room_id: '',
    stage: MILESTONE_STAGES[0],
    caption: '',
    specsInput: '',
    cost: '',
    contractor: '',
  })
  const [savingEdit, setSavingEdit] = useState(false)

  useEffect(() => {
    const loadTimeline = async () => {
      const { data: auth } = await supabase.auth.getUser()
      if (!auth.user) return router.push('/login')

      const { data: roomData } = await supabase
        .from('rooms')
        .select('id, name')
        .order('created_at', { ascending: true })
      if (roomData) setRooms(roomData)

      const { data: postData } = await supabase
        .from('posts')
        .select('*')
        .order('created_at', { ascending: false })
      if (postData) setPosts(postData)

      setLoading(false)
    }
    loadTimeline()
  }, [router])

  const openEditModal = (post: PostItem) => {
    setEditingPost(post)
    setEditForm({
      room_id: post.room_id || '',
      stage: post.stage || MILESTONE_STAGES[0],
      caption: post.caption || '',
      specsInput: Array.isArray(post.specs) ? post.specs.join(', ') : '',
      cost: post.cost !== null && post.cost !== undefined ? String(post.cost) : '',
      contractor: post.contractor || '',
    })
  }

  const handleSaveEdit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!editingPost) return
    setSavingEdit(true)

    const parsedSpecs = editForm.specsInput
      .split(',')
      .map((s) => s.trim())
      .filter(Boolean)

    const updates = {
      room_id: editForm.room_id || null,
      stage: editForm.stage,
      caption: editForm.caption.trim() || null,
      specs: parsedSpecs.length > 0 ? parsedSpecs : null,
      cost: editForm.cost !== '' ? Number(editForm.cost) : null,
      contractor: editForm.contractor.trim() || null,
    }

    const { data, error } = await supabase
      .from('posts')
      .update(updates)
      .eq('id', editingPost.id)
      .select()
      .single()

    setSavingEdit(false)
    if (error) {
      alert('Error updating post: ' + error.message)
    } else if (data) {
      setPosts((prev) => prev.map((p) => (p.id === data.id ? data : p)))
      setEditingPost(null)
    }
  }

  const handleDeletePost = async () => {
    if (!editingPost) return
    if (!confirm('Delete this renovation update permanently?')) return

    const { error } = await supabase.from('posts').delete().eq('id', editingPost.id)
    if (error) {
      alert('Error deleting post: ' + error.message)
    } else {
      setPosts((prev) => prev.filter((p) => p.id !== editingPost.id))
      setEditingPost(null)
    }
  }

  const getRoomName = (roomId: string | null) => {
    if (!roomId) return 'Whole House / Exterior'
    return rooms.find((r) => r.id === roomId)?.name || 'Unassigned Room'
  }

  const filteredPosts =
    selectedRoomFilter === 'all'
      ? posts
      : selectedRoomFilter === 'whole'
      ? posts.filter((p) => !p.room_id)
      : posts.filter((p) => p.room_id === selectedRoomFilter)

  if (loading) return <div className="p-6 text-xs text-stone-500">Loading timeline...</div>

  return (
    <div className="p-5 space-y-5">
      <div className="flex items-center justify-between">
        <div>
          <p className="text-[11px] uppercase tracking-wider font-semibold text-[#5C6E52]">Renovation Log</p>
          <h1 className="text-xl font-semibold text-stone-900">House Timeline</h1>
        </div>
        <Link
          href="/upload"
          className="text-xs font-medium bg-[#7A8F6D] text-white px-3 py-1.5 rounded-full"
        >
          + Log Update
        </Link>
      </div>

      {/* Room Filter Pills */}
      <div className="flex gap-2 overflow-x-auto pb-1">
        <button
          onClick={() => setSelectedRoomFilter('all')}
          className={`shrink-0 px-3 py-1 rounded-full text-xs font-medium border ${
            selectedRoomFilter === 'all'
              ? 'bg-[#3A3D38] text-white border-[#3A3D38]'
              : 'bg-white text-stone-700 border-[#D8D3C8]'
          }`}
        >
          All ({posts.length})
        </button>
        {rooms.map((r) => (
          <button
            key={r.id}
            onClick={() => setSelectedRoomFilter(r.id)}
            className={`shrink-0 px-3 py-1 rounded-full text-xs font-medium border ${
              selectedRoomFilter === r.id
                ? 'bg-[#3A3D38] text-white border-[#3A3D38]'
                : 'bg-white text-stone-700 border-[#D8D3C8]'
            }`}
          >
            {r.name}
          </button>
        ))}
        <button
          onClick={() => setSelectedRoomFilter('whole')}
          className={`shrink-0 px-3 py-1 rounded-full text-xs font-medium border ${
            selectedRoomFilter === 'whole'
              ? 'bg-[#3A3D38] text-white border-[#3A3D38]'
              : 'bg-white text-stone-700 border-[#D8D3C8]'
          }`}
        >
          Whole House
        </button>
      </div>

      {/* Posts Feed */}
      {filteredPosts.length === 0 ? (
        <div className="p-6 rounded-2xl bg-[#EAE6DF] border border-[#DCD6CA] text-center space-y-2">
          <p className="text-xs text-stone-700">No updates logged for this view yet.</p>
          <Link href="/upload" className="inline-block text-xs font-semibold text-[#4B5E42] underline">
            Log your first milestone →
          </Link>
        </div>
      ) : (
        <div className="space-y-4">
          {filteredPosts.map((post) => (
            <article
              key={post.id}
              className="rounded-2xl bg-white border border-[#DCD6CA] overflow-hidden shadow-2xs"
            >
              {/* Card Header */}
              <div className="px-4 py-3 flex items-center justify-between border-b border-[#F0EDE6]">
                <div className="space-y-0.5">
                  <div className="flex items-center gap-2">
                    {post.room_id ? (
                      <Link
                        href={`/rooms/${post.room_id}`}
                        className="text-xs font-semibold text-stone-900 hover:text-[#5C6E52]"
                      >
                        {getRoomName(post.room_id)}
                      </Link>
                    ) : (
                      <span className="text-xs font-semibold text-stone-900">Whole House / Exterior</span>
                    )}
                    {post.stage && (
                      <span className="text-[10px] font-medium px-2 py-0.5 rounded-full bg-[#DCE6D5] text-[#2E3B27]">
                        {post.stage}
                      </span>
                    )}
                  </div>
                  <p className="text-[10px] text-stone-400">
                    {new Date(post.created_at).toLocaleDateString('en-GB', {
                      day: 'numeric',
                      month: 'short',
                      year: 'numeric',
                    })}
                    {post.contractor ? ` · ${post.contractor}` : ''}
                  </p>
                </div>

                <button
                  onClick={() => openEditModal(post)}
                  className="text-xs font-medium text-stone-500 hover:text-stone-900 underline px-2 py-1"
                >
                  Edit
                </button>

                      £{Number(post.cost).toLocaleString('en-GB')}
                    </span>
                  )}
                </div>
              </div>
            </article>
          ))}
        </div>
      )}

        >
          <div
            className="w-full max-w-sm rounded-2xl bg-[#F5F3EE] border border-[#DCD6CA] p-5 space-y-4 shadow-xl max-h-[90vh] overflow-y-auto"
            onClick={(e) => e.stopPropagation()}
          >
            <div className="flex items-center justify-between">
              <div>
                <p className="text-[10px] uppercase tracking-wider text-[#5C6E52] font-semibold">Edit Log Entry</p>

'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { supabase } from '@/lib/supabase'

export default function LoginPage() {
  const router = useRouter()
  const [isSignUp, setIsSignUp] = useState(false)
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [loading, setLoading] = useState(false)
  const [errorMsg, setErrorMsg] = useState<string | null>(null)

  const handleAuth = async (e: React.FormEvent) => {
    e.preventDefault()
    setLoading(true)
    setErrorMsg(null)

    const { error } = isSignUp
      ? await supabase.auth.signUp({ email, password })
      : await supabase.auth.signInWithPassword({ email, password })

    setLoading(false)

    if (error) {
      setErrorMsg(error.message)
    } else {
      router.push('/upload')
      router.refresh()
    }
  }

  return (
    <div className="p-6 space-y-6 max-w-md mx-auto">
      <div className="space-y-1">
        <p className="text-xs tracking-wider text-stone-400 uppercase font-medium">Account Access</p>
        <h1 className="text-xl font-medium text-stone-900">
          {isSignUp ? 'Create your Passport' : 'Sign in to homewallr'}
        </h1>
      </div>

      {errorMsg && (
        <div className="p-3 bg-red-50 border border-red-100 rounded-xl text-xs text-red-600">
          {errorMsg}
        </div>
      )}

      <form onSubmit={handleAuth} className="space-y-4">
        <div className="space-y-1.5">
          <label className="text-xs font-medium text-stone-700">Email</label>
          <input
            type="email"
            required
            placeholder="you@example.com"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            className="w-full px-3 py-2 text-sm text-stone-900 placeholder:text-stone-400 bg-stone-50 border border-stone-200 rounded-xl focus:outline-none focus:border-stone-400"
          />
        </div>

        <div className="space-y-1.5">
          <label className="text-xs font-medium text-stone-700">Password</label>
          <input
            type="password"
            required
            minLength={6}
            placeholder="••••••••"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            className="w-full px-3 py-2 text-sm text-stone-900 placeholder:text-stone-400 bg-stone-50 border border-stone-200 rounded-xl focus:outline-none focus:border-stone-400"
          />
        </div>

        <button
          type="submit"
          disabled={loading}
          className="w-full py-2.5 bg-stone-900 text-white text-sm font-medium rounded-xl hover:bg-stone-800 transition-colors disabled:opacity-50"
        >
          {loading ? 'Please wait...' : isSignUp ? 'Create Account' : 'Sign In'}
        </button>
      </form>

      <div className="text-center pt-2">
        <button
          type="button"
          onClick={() => {
            setIsSignUp(!isSignUp)
            setErrorMsg(null)
          }}
          className="text-xs text-stone-500 hover:text-stone-900 underline"
        >
          {isSignUp ? 'Already have an account? Sign in' : "Don't have an account? Sign up"}
        </button>
      </div>
    </div>

'use client'

import { useState, useRef } from 'react'
import { useRouter } from 'next/navigation'
import { supabase } from '@/lib/supabase'

export type WallSide = 'top' | 'bottom' | 'left' | 'right'
export type FeatureType = 'door' | 'window' | 'chimney' | 'stairs'

export interface RoomFeature {
  id: string
  type: FeatureType
  wall: WallSide
  offset: number // percentage along wall (10 to 90)
  size_m?: number // width in meters (e.g. 0.85m door, 1.4m window)
}

export interface PlannerRoom {
  id: string
  name: string
  status: string
  progress: number
  floor_level: string | null
  width_m: number | null
  length_m: number | null
  plan_x: number | null
  plan_y: number | null
  features?: RoomFeature[] | null
}

interface Props {
  rooms: PlannerRoom[]
  onRoomsChange: (rooms: PlannerRoom[]) => void
}

const SCALE = 28 // 28px = 1m
const SNAP = 7   // 7px = 0.25m

const DEFAULT_SIZES: Record<FeatureType, number> = {
  door: 0.85,
  window: 1.2,
  chimney: 1.1,
  stairs: 0.9,
}

export default function FloorPlanBuilder({ rooms, onRoomsChange }: Props) {
  const router = useRouter()
  const canvasRef = useRef<HTMLDivElement>(null)

  const [floor, setFloor] = useState<'Ground Floor' | 'First Floor'>('Ground Floor')
  const [viewMode, setViewMode] = useState<'2d' | '3d'>('2d')
  const [editMode, setEditMode] = useState(false)
  const [zoom, setZoom] = useState(1)

  const [selectedId, setSelectedId] = useState<string | null>(null)
  const [selectedFeatureId, setSelectedFeatureId] = useState<string | null>(null)
  const [activeWall, setActiveWall] = useState<WallSide>('top')

  const [interaction, setInteraction] = useState<{
    type: 'move' | 'resize-right' | 'resize-bottom' | 'resize-left' | 'resize-top'
    roomId: string
    startX: number
    startY: number
    origX: number
    origY: number
    origW: number
    origL: number
  } | null>(null)

  const floorRooms = rooms.filter((r) => (r.floor_level || 'Ground Floor') === floor)
  const selectedRoom = rooms.find((r) => r.id === selectedId) || null
  const selectedFeature = selectedRoom?.features?.find((f) => f.id === selectedFeatureId) || null
  const totalArea = floorRooms.reduce((s, r) => s + (Number(r.width_m) || 3.5) * (Number(r.length_m) || 3), 0)

  const startInteraction = (
    e: React.PointerEvent,
    room: PlannerRoom,
    type: 'move' | 'resize-right' | 'resize-bottom' | 'resize-left' | 'resize-top'
  ) => {
    e.stopPropagation()
    if (selectedId !== room.id) setSelectedFeatureId(null)
    setSelectedId(room.id)
    if (!editMode || viewMode === '3d') return

    setInteraction({
      type,
      roomId: room.id,
      startX: e.clientX,
      startY: e.clientY,
      origX: Number(room.plan_x) || 20,
      origY: Number(room.plan_y) || 20,
      origW: Number(room.width_m) || 3.5,
      origL: Number(room.length_m) || 3.0,
    })
  }

  const handlePointerMove = (e: React.PointerEvent) => {
    if (!interaction) return
    const dx = Math.round((e.clientX - interaction.startX) / zoom / SNAP) * SNAP
    const dy = Math.round((e.clientY - interaction.startY) / zoom / SNAP) * SNAP
    const dMetersX = dx / SCALE
    const dMetersY = dy / SCALE

    onRoomsChange(
      rooms.map((r) => {
        if (r.id !== interaction.roomId) return r
        if (interaction.type === 'move') {
          return { ...r, plan_x: Math.max(0, interaction.origX + dx), plan_y: Math.max(0, interaction.origY + dy) }
        }
        if (interaction.type === 'resize-right') {
          return { ...r, width_m: Math.max(1.25, Number((interaction.origW + dMetersX).toFixed(2))) }
        }
        if (interaction.type === 'resize-bottom') {
          return { ...r, length_m: Math.max(1.25, Number((interaction.origL + dMetersY).toFixed(2))) }
        }
        if (interaction.type === 'resize-left') {
          const newW = Math.max(1.25, Number((interaction.origW - dMetersX).toFixed(2)))
          const actualDx = (interaction.origW - newW) * SCALE
          return { ...r, width_m: newW, plan_x: Math.max(0, interaction.origX + actualDx) }
        }
        if (interaction.type === 'resize-top') {
          const newL = Math.max(1.25, Number((interaction.origL - dMetersY).toFixed(2)))
          const actualDy = (interaction.origL - newL) * SCALE
          return { ...r, length_m: newL, plan_y: Math.max(0, interaction.origY + actualDy) }
        }
        return r
      })
    )
  }

  const handlePointerUp = async () => {
    if (!interaction) return
    const updated = rooms.find((r) => r.id === interaction.roomId)
    setInteraction(null)
    if (updated) {
      await supabase
        .from('rooms')
        .update({
          plan_x: updated.plan_x,
          plan_y: updated.plan_y,
          width_m: updated.width_m,
          length_m: updated.length_m,
        })
        .eq('id', updated.id)
    }
  }

    if (!selectedRoom) return
    const current = selectedRoom.features || []
    const newId = `${Date.now()}`
    const nextFeatures: RoomFeature[] = [
      ...current,
      { id: newId, type, wall: activeWall, offset: 50, size_m: DEFAULT_SIZES[type] },
    ]

    setSelectedFeatureId(newId)
    onRoomsChange(rooms.map((r) => (r.id === selectedRoom.id ? { ...r, features: nextFeatures } : r)))
    await supabase.from('rooms').update({ features: nextFeatures }).eq('id', selectedRoom.id)
  }

  const updateSelectedFeature = async (patch: Partial<RoomFeature>) => {
    if (!selectedRoom || !selectedFeatureId) return
    const nextFeatures = (selectedRoom.features || []).map((f) =>
      f.id === selectedFeatureId ? { ...f, ...patch } : f
    )
    onRoomsChange(rooms.map((r) => (r.id === selectedRoom.id ? { ...r, features: nextFeatures } : r)))
    await supabase.from('rooms').update({ features: nextFeatures }).eq('id', selectedRoom.id)
  }

  const deleteSelectedFeature = async () => {
    if (!selectedRoom || !selectedFeatureId) return
    const nextFeatures = (selectedRoom.features || []).filter((f) => f.id !== selectedFeatureId)
    setSelectedFeatureId(null)
    onRoomsChange(rooms.map((r) => (r.id === selectedRoom.id ? { ...r, features: nextFeatures } : r)))
    await supabase.from('rooms').update({ features: nextFeatures }).eq('id', selectedRoom.id)
  }

  return (
    <div className="space-y-3">
      {/* Header & View/Edit Controls */}
      <div className="flex justify-between items-center gap-2">
        <div>
          <h2 className="text-xs tracking-wider uppercase font-semibold text-stone-700">Architectural Plan</h2>
          <p className="text-[11px] text-stone-500">{floor} · {totalArea.toFixed(1)} m²</p>
        </div>

        <div className="flex items-center gap-1.5">
          {/* Zoom Controls */}
          <div className="flex items-center bg-white border border-[#DCD6CA] rounded-full px-1.5 py-0.5 text-xs">
            <button
              type="button"
              onClick={() => setZoom((z) => Math.max(0.6, Number((z - 0.15).toFixed(2))))}
              className="px-1.5 font-bold text-stone-700 hover:text-stone-950"
              title="Zoom Out"
            >
              −
            </button>
            <button
              type="button"
              onClick={() => setZoom(1)}
              className="px-1 text-[10px] font-medium text-stone-500"
              title="Reset Zoom"
            >
              {Math.round(zoom * 100)}%
            </button>
            <button
              type="button"
              onClick={() => setZoom((z) => Math.min(1.8, Number((z + 0.15).toFixed(2))))}
              className="px-1.5 font-bold text-stone-700 hover:text-stone-950"
              title="Zoom In"
            >
              +
            </button>
          </div>

          <button
            onClick={() => {
              setViewMode(viewMode === '2d' ? '3d' : '2d')
              setEditMode(false)
              setSelectedFeatureId(null)
            }}
            className={`text-xs font-medium px-2.5 py-1.5 rounded-full border transition-colors ${

import { useState, useEffect, useRef } from 'react'
import Link from 'next/link'
import { supabase } from '@/lib/supabase'

export interface FloorPlanRoom {
  id: string
  name: string
  floor_level?: string | null
  width_m?: number | null
  length_m?: number | null
  status: string
  progress: number
}

interface AIFloorPlanProps {
  userId: string
  rooms: FloorPlanRoom[]
  plan2dUrl?: string | null
  plan3dUrl?: string | null
  onRoomsUpdated: () => Promise<void>
}

const GENERATION_STEPS = [
  'Reading room dimensions & stripping watermarks...',
  'Tracing outer walls, stairs & chimney breasts...',
  'Placing doorway swings, windows & room labels...',
  'Styling blueprint & syncing rooms...',
]

async function dataUrlToBlob(dataUrl: string): Promise<Blob> {
  const res = await fetch(dataUrl)
  return res.blob()
}

export default function AIFloorPlan({
  userId,
  rooms,
  plan2dUrl,
  onRoomsUpdated,
}: AIFloorPlanProps) {
  const fileInputRef = useRef<HTMLInputElement | null>(null)
  const [generating, setGenerating] = useState(false)
  const [stepIndex, setStepIndex] = useState(0)
  const [elapsedSeconds, setElapsedSeconds] = useState(0)
  const [img2D, setImg2D] = useState<string | null>(plan2dUrl || null)
  const [fullscreenOpen, setFullscreenOpen] = useState(false)

  useEffect(() => {
    if (!generating) {
      setStepIndex(0)
      setElapsedSeconds(0)
      return
    }

    const timer = setInterval(() => {
      setElapsedSeconds((prev) => prev + 1)
    }, 1000)

    const stepper = setInterval(() => {
      setStepIndex((prev) => (prev + 1 < GENERATION_STEPS.length ? prev + 1 : prev))
    }, 9000)

    return () => {
      clearInterval(timer)
      clearInterval(stepper)
    }
  }, [generating])

  const handleUploadAndGenerate = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    setGenerating(true)

    try {
      const formData = new FormData()
      formData.append('file', file)

      const res = await fetch('/api/parse-floorplan', {
        method: 'POST',
        body: formData,
      })
      const data = await res.json()

      if (!res.ok) {
        alert(data.error || 'Could not generate floor plan.')
        setGenerating(false)
        return
      }

      let saved2dUrl = data.image2D || img2D

      if (data.image2D) {
        setImg2D(data.image2D)
        const blob2d = await dataUrlToBlob(data.image2D)
        const path2d = `floorplans/${userId}-2d-${Date.now()}.svg`
        const { error: err2d } = await supabase.storage
          .from('post-images')
          .upload(path2d, blob2d, { contentType: 'image/svg+xml' })
        if (!err2d) {
          saved2dUrl = supabase.storage.from('post-images').getPublicUrl(path2d).data.publicUrl
          setImg2D(saved2dUrl)
        }
      }

      await supabase.from('houses').upsert(
        {
          user_id: userId,
          plan_2d_url: saved2dUrl,
        },
        { onConflict: 'user_id' }
      )

      if (Array.isArray(data.rooms)) {
        for (const r of data.rooms) {
          const exists = rooms.some(
            (existing) => existing.name.trim().toLowerCase() === String(r.name || '').trim().toLowerCase()
          )
          if (!exists && r.name) {
            await supabase.from('rooms').insert([
              {
                user_id: userId,
                name: r.name,
                floor_level: r.floor_level || 'Ground Floor',
                width_m: Number(r.width_m) || 3.5,
                length_m: Number(r.length_m) || 3.0,
                status: 'Planned',
                progress: 0,
              },
            ])
          }
        }
        await onRoomsUpdated()
      }
    } catch {
      alert('Something went wrong generating your floor plan.')
    } finally {
      setGenerating(false)
      if (fileInputRef.current) fileInputRef.current.value = ''
    }
  }

  const progressPercent = Math.min(92, Math.round((elapsedSeconds / 40) * 92) + 8)

  return (
    <div className="rounded-2xl bg-[#FAF8F4] border border-[#DCD6CA] p-4 space-y-3.5">
      {/* Top Header & Upload Trigger */}
      <div className="flex items-center justify-between gap-2">
        <div>
          <p className="text-[10px] uppercase tracking-wider font-semibold text-[#5C6E52]">
            House Floor Plan
          </p>
          <h3 className="text-sm font-semibold text-stone-900">Architectural Blueprint</h3>
        </div>

        <div className="flex items-center gap-2">
          {img2D && !generating && (
            <button
              type="button"
              onClick={() => setFullscreenOpen(true)}
              className="text-[11px] font-medium text-[#5C6E52] underline px-1"
            >
              Expand ⤢
            </button>
          )}

          <label className="cursor-pointer inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-[#7A8F6D] hover:bg-[#687C5C] text-white text-xs font-medium transition-colors">
            <span>{generating ? `Drafting (${elapsedSeconds}s)...` : img2D ? '↻ Re-scan Plan' : '✨ Generate Plan'}</span>
            <input
              ref={fileInputRef}
              type="file"
              accept="image/*"
              disabled={generating}
              onChange={handleUploadAndGenerate}
              className="hidden"
            />
          </label>
        </div>
      </div>

      {/* Main Generated Floor Plan Vector */}
      {generating ? (
        <div className="rounded-xl border border-[#C7D2BF] bg-[#E7EAE3]/70 py-10 px-5 text-center space-y-3">
          <p className="text-xs font-semibold text-[#2E3B27]">
            {GENERATION_STEPS[stepIndex]}
          </p>

          <div className="w-full max-w-xs mx-auto h-2 bg-white rounded-full overflow-hidden border border-[#C7D2BF]">
            <div
              className="h-full bg-[#7A8F6D] rounded-full transition-all duration-1000"
              style={{ width: `${progressPercent}%` }}
            />
          </div>

          <p className="text-[11px] text-stone-600 leading-relaxed">
            Redrawing your floor plan into a clean architectural blueprint usually takes <strong>30 to 60 seconds</strong> — please keep this screen open.
          </p>
        </div>
      ) : img2D ? (
        <button
          type="button"
          onClick={() => setFullscreenOpen(true)}
          className="block w-full rounded-xl overflow-hidden border border-[#DCD6CA] bg-[#F5F3EE] focus:outline-none"
        >
          <img
            src={img2D}
            alt="Styled 2D Architectural Floor Plan"
            className="w-full max-h-80 object-contain mx-auto"
          />
        </button>
      ) : (
        <div className="rounded-xl border border-dashed border-[#B9C9AE] bg-[#F5F3EE] p-6 text-center space-y-1.5">
          <p className="text-xs font-medium text-stone-800">Upload an estate agent plan or sketch</p>

      {rooms.length > 0 && (

  'Polishing architectural vector & saving to your house...',
]

const FloorPlanContext = createContext<FloorPlanJobState | undefined>(undefined)

async function dataUrlToBlob(dataUrl: string): Promise<Blob> {
  const res = await fetch(dataUrl)
  return res.blob()
}

export function FloorPlanProvider({ children }: { children: React.ReactNode }) {
  const router = useRouter()
  const [isGenerating, setIsGenerating] = useState(false)
  const [elapsedSeconds, setElapsedSeconds] = useState(0)
  const [stepIndex, setStepIndex] = useState(0)
  const [latestPlanUrl, setLatestPlanUrl] = useState<string | null>(null)
  const [completedNotification, setCompletedNotification] = useState(false)
  const [errorMessage, setErrorMessage] = useState<string | null>(null)

  useEffect(() => {
    if (!isGenerating) {
      setElapsedSeconds(0)
      setStepIndex(0)
      return
    }

    const timer = setInterval(() => {
      setElapsedSeconds((prev) => prev + 1)
    }, 1000)

    const stepper = setInterval(() => {
      setStepIndex((prev) => (prev + 1 < GENERATION_STEPS.length ? prev + 1 : prev))
    }, 10000)

    return () => {
      clearInterval(timer)
      clearInterval(stepper)
    }
  }, [isGenerating])

  const startGeneration = useCallback(
    async (file: File, userId: string, existingRoomNames: string[]) => {
      if (isGenerating) return

      if (typeof window !== 'undefined' && 'Notification' in window && Notification.permission === 'default') {
        Notification.requestPermission().catch(() => {})
      }

      setIsGenerating(true)
      setCompletedNotification(false)
      setErrorMessage(null)

      try {
        const formData = new FormData()
        formData.append('file', file)

        const res = await fetch('/api/parse-floorplan', {
          method: 'POST',
          body: formData,
        })
        const data = await res.json()

        if (!res.ok) {
          setErrorMessage(data.error || 'Could not generate floor plan.')
          setIsGenerating(false)
          return
        }

        let saved2dUrl: string | null = data.image2D || null

        if (data.image2D) {
          const blob2d = await dataUrlToBlob(data.image2D)
          const path2d = `floorplans/${userId}-2d-${Date.now()}.svg`
          const { error: err2d } = await supabase.storage
            .from('post-images')
            .upload(path2d, blob2d, { contentType: 'image/svg+xml' })

          if (!err2d) {
            saved2dUrl = supabase.storage.from('post-images').getPublicUrl(path2d).data.publicUrl
          }
        }

        if (saved2dUrl) {
          await supabase.from('houses').upsert(
            {
              user_id: userId,
              plan_2d_url: saved2dUrl,
            },
            { onConflict: 'user_id' }
          )
          setLatestPlanUrl(saved2dUrl)
        }

        if (Array.isArray(data.rooms)) {
          const normalizedExisting = new Set(existingRoomNames.map((n) => n.trim().toLowerCase()))
          for (const r of data.rooms) {
            const cleanName = String(r.name || '').trim()
            if (cleanName && !normalizedExisting.has(cleanName.toLowerCase())) {
              await supabase.from('rooms').insert([
                {
                  user_id: userId,
                  name: cleanName,
                  floor_level: r.floor_level || 'Ground Floor',
                  width_m: Number(r.width_m) || 3.5,
                  length_m: Number(r.length_m) || 3.0,
                  status: 'Planned',
                  progress: 0,
                },
              ])
              normalizedExisting.add(cleanName.toLowerCase())
            }
          }
        }

        setCompletedNotification(true)

        if (typeof window !== 'undefined' && 'Notification' in window && Notification.permission === 'granted') {
          new Notification('HomeWallr · Floor Plan Ready', {
            body: 'Your architectural floor plan has finished generating and is saved to your profile.',
          })
        }
      } catch {
        setErrorMessage('Something went wrong while generating your floor plan.')
      } finally {
        setIsGenerating(false)
      }
    },
    [isGenerating]
  )

  const dismissNotification = () => setCompletedNotification(false)
  const dismissError = () => setErrorMessage(null)

  const formatElapsed = (secs: number) => {
    const mins = Math.floor(secs / 60)
    const rem = secs % 60
    return mins > 0 ? `${mins}m ${rem}s` : `${rem}s`
  }

  return (
    <FloorPlanContext.Provider
      value={{
        isGenerating,
        elapsedSeconds,
        stepText: GENERATION_STEPS[stepIndex],
        latestPlanUrl,
        completedNotification,
        errorMessage,
        startGeneration,
        dismissNotification,
        dismissError,
      }}
    >
      {children}

      {/* Global Floating Status Pill (Visible across all pages while generating) */}
      {isGenerating && (
        <div className="fixed top-3 left-1/2 -translate-x-1/2 z-50 w-[92%] max-w-sm rounded-2xl bg-[#263022] text-white px-4 py-3 shadow-xl border border-[#4D6144] flex items-center justify-between gap-3">
          <div className="min-w-0 space-y-0.5">
            <p className="text-xs font-semibold flex items-center gap-2">
              <span className="inline-block w-2 h-2 rounded-full bg-[#A3C293] animate-pulse" />
              Drafting Floor Plan ({formatElapsed(elapsedSeconds)})
            </p>
            <p className="text-[10px] text-stone-300 truncate">
              Feel free to browse HomeWallr while we draft it
            </p>
          </div>
          <span className="shrink-0 text-[10px] font-medium px-2 py-1 rounded-full bg-white/10 text-[#CFE0C5]">
            Background
          </span>
        </div>
      )}

      {/* Global Completion Toast Notification */}
      {completedNotification && !isGenerating && (
        <div className="fixed top-3 left-1/2 -translate-x-1/2 z-50 w-[92%] max-w-sm rounded-2xl bg-[#7A8F6D] text-white px-4 py-3.5 shadow-2xl border border-[#A5B999] flex items-center justify-between gap-3">
          <div className="min-w-0">
            <p className="text-xs font-semibold">✨ Architectural Floor Plan Ready!</p>
            <p className="text-[11px] text-white/90">Saved to your house profile & rooms synced.</p>
          </div>
          <div className="flex items-center gap-1.5 shrink-0">
            <button
              type="button"
              onClick={() => {
                dismissNotification()
                router.push('/profile')
              }}
              className="px-3 py-1.5 rounded-full bg-white text-[#263022] text-xs font-semibold shadow-2xs"
            >
              View →
            </button>
            <button
              type="button"
              onClick={dismissNotification}
              aria-label="Dismiss notification"
              className="w-6 h-6 rounded-full bg-black/15 text-white text-xs flex items-center justify-center"
            >
              ×
            </button>
          </div>
        </div>
      )}

      {/* Global Error Toast */}
      {errorMessage && !isGenerating && (
        <div className="fixed top-3 left-1/2 -translate-x-1/2 z-50 w-[92%] max-w-sm rounded-2xl bg-red-900 text-white px-4 py-3 shadow-xl flex items-center justify-between gap-3">
          <p className="text-xs">{errorMessage}</p>
          <button
            type="button"
            onClick={dismissError}
            className="w-6 h-6 rounded-full bg-white/15 text-white text-xs flex items-center justify-center shrink-0"
          >
            ×
          </button>
        </div>
      )}
    </FloorPlanContext.Provider>
  )
}

export function useFloorPlanJob() {
  const ctx = useContext(FloorPlanContext)
  if (!ctx) {
    throw new Error('useFloorPlanJob must be used inside FloorPlanProvider')
  }
  return ctx
}
